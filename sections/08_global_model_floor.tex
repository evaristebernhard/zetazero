\section{Positive reference model and the true \texorpdfstring{one-$\chi$}{one-chi} defect}
\label{sec:model-floor}

Let $\vartheta_{\rm c}$ denote the fixed coordinate multiplier implementing the
stable-core restriction on each packet leg, and put
$\vartheta_{\rm e}=1-\vartheta_{\rm c}$.  For an exact right-edge source $D$
and leg multipliers $a,b$, define the Hermitian operator
\begin{equation}
\langle\mathfrak R_T[D;a,b]v,w\rangle
:=\frac1{2\pi i}\int_{\Gamma_c^+}q(s)^2D(s)
 \mathcal B_{v,w}^{a,b}(s)\,ds
+\overline{\frac1{2\pi i}\int_{\Gamma_c^+}q(s)^2D(s)
 \mathcal B_{w,v}^{b,a}(s)\,ds}.
\label{eq:exact-right-edge-operator}
\end{equation}
Here
\[
\mathcal B_{v,w}^{a,b}(s):=B_w^{(b)}(z(s))
 \bigl(B_v^{(a)}\bigr)^\#(z(s))
\]
where $B_v^{(a)}$ denotes the packet synthesis after applying the multiplier
$a$ to its leg.  Thus this is the exact packet polarization after applying the
indicated multipliers to the two legs; no stationary expansion is part of this
definition.  Put
\begin{align}
A_T^{\rm ar}&:=\mathfrak R_T[\mathcal C H_{\rm ar};1,1],\notag\\
A_T^{\rm true}&:=\mathfrak R_T[\mathcal C H_{\rm ar};
 \vartheta_{\rm c},\vartheta_{\rm c}],                      \label{eq:true-logonly-operator}\\
L_T^{\rm geom}&:=A_T^{\rm ar}-A_T^{\rm true}.               \label{eq:exact-geometric-leakage}
\end{align}
Thus $L_T^{\rm geom}$ is exactly the sum of the three terms with at least one
$\vartheta_{\rm e}$ leg.  The physical additive carrier $e(-ne^d)$ and the
complete direct residual are retained inside $A_T^{\rm true}$.  No claim is
made that the untwisted scalar HLZ diagonal identifies this operator with the
comparison model below.

Let $C(x):=G_{\log}(1-x)$ on the retained interior lag interval.  If $q$ is
mean zero and
\[
Q(x):=\int_0^xq(t)\,dt,
\]
then two integrations by parts give the exact primitive identity
\begin{equation}
\boxed{
\langle K_Cq,q\rangle
=-2C'(0+)\|Q\|_2^2
-\iint C''(|x-y|)Q(x)\overline{Q(y)}\,dx\,dy.
}
\label{eq:primitive-identity}
\end{equation}
Define
\[
Q_\triangle(q):=2\|Q\|_2^2.
\]

On the central flat plateau translate the physical Fourier coordinate by its
fixed left endpoint $u_0$.  The flat packet synthesis is
\begin{equation}
\boxed{
(\mathcal U_T^{\rm flat}v)(u)
=L_{\rm pkt}^{-1/2}\sum_\nu
 e^{i\tau_\nu u_0}v_\nu e^{i\tau_\nu u}.
}
\label{eq:explicit-flat-analysis}
\end{equation}
Let $C_T$ be the stable-core restriction, put
\[
R_T^{\rm core}:=\frac{W_T}{L},
\qquad
\mathcal U_T^{\rm core}:=C_T\mathcal U_T^{\rm flat},
\]
and normalize by $r=u/L$:
\begin{equation}
\boxed{
(\widehat{\mathcal U}_T^{\rm core}v)(r)
:=L^{1/2}(\mathcal U_T^{\rm core}v)(Lr),
\qquad 0\le r\le R_T^{\rm core}.
}
\label{eq:normalized-core-analysis}
\end{equation}
The global reference form is the pullback, through this same core analysis,
of $Q_\triangle$.  Equivalently
\begin{equation}
\boxed{
\Gfr_T=(\mathcal W_T^{\rm glob})^*\mathcal W_T^{\rm glob},
}
\label{eq:global-reference-Gram}
\end{equation}
where $\mathcal W_T^{\rm glob}$ adjoins the primitive on
$0\le u\le W_T$ and the normalization $\sqrt2/L$.

For the conditional comparison, define the positive reference kernel directly
by
\[
K_{G_{\log}}(r,s)=G_{\log}(1-|r-s|),
\]
with
\begin{equation}
\boxed{
G_{\log}(\rho)
=-\frac76\rho^4+2\rho^3-\rho^2
+\frac{e^2}{8}\left[
(2\rho^2-4\rho+3)\cosh(2\rho)-4\rho^2+4\rho-3
\right].
}
\label{eq:Glog-closed}
\end{equation}
Equivalently, its hyperbolic contribution is
$\frac{e^2}{4}\{(2\rho^2-4\rho+3)\sinh^2\rho-\rho^2\}$.
This is the explicit reference profile used in the spectral comparison; it is
not obtained by identifying the true packet operator with an untwisted scalar
lag kernel.

For later certification we use the elementary rational enclosure
\begin{equation}
\sum_{j=0}^{m}\frac{x^j}{j!}
\le e^x\le
\sum_{j=0}^{m}\frac{x^j}{j!}
+\frac{x^{m+1}}{(m+1)!}\frac1{1-x/(m+2)}
\qquad(0\le x\le4,\ m+2>x).
\label{eq:exp-rational-bound}
\end{equation}

\begin{proposition}[Certified log-only spectral floor]
\label{prop:model-floor}
On every mean-zero stable core fibre,
\begin{equation}
\boxed{
K_{G_{\log}}\succeq\kappa_*\,Q_\triangle.
}
\label{eq:model-floor}
\end{equation}
Consequently, for the reference packet pullback
\[
P_T:=A_T^{G_{\log}},
\]
one has
\begin{equation}
\boxed{P_T\succeq\kappa_*\,\Gfr_T.}
\label{eq:logonly-model-floor}
\end{equation}
This is a statement about the reference model $P_T$; it does not identify
$P_T$ with $A_T^{\rm true}$.
\end{proposition}

\begin{proof}
Extend a core input by zero to $[0,1]$.  Put
\[
g_{\log}(r):=G_{\log}''(r).
\]
Direct differentiation of \eqref{eq:Glog-closed} gives
\begin{equation}
\boxed{
g_{\log}(r)=-14r^2+12r-2
+e^2\left[(r^2-2r+2)\cosh(2r)
 +2(r-1)\sinh(2r)-1\right].
}
\label{eq:glog-second}
\end{equation}
The cusp coefficient in \eqref{eq:primitive-identity} is
\begin{equation}
\boxed{
a_{\log}=2G_{\log}'(1)
=-\frac43+\frac{e^2}{2}(\sinh2-2)>4.6770.}
\label{eq:logonly-cusp}
\end{equation}

Exact rational interval arithmetic using \eqref{eq:exp-rational-bound} on a
mesh of width $10^{-4}$ gives
\begin{equation}
\begin{gathered}
 g_{\log}''(r)>21.80\quad(0\le r\le1),\\
 g_{\log}(0.2483)>0>g_{\log}(0.2484),\qquad
 g_{\log}(0.6464)<0<g_{\log}(0.6465),\\
 g_{\log}'(x)+g_{\log}'(1-x)>5.70\quad(0\le x\le\tfrac12).
\end{gathered}
\label{eq:glog-root-certificate}
\end{equation}
Hence $g_{\log}$ is strictly convex, has exactly two roots
\[
\alpha\in(0.2483,0.2484),\qquad
\beta\in(0.6464,0.6465),
\]
and has signs $+,-,+$ on the three complementary intervals.

Let
\[
M_{\log}(x):=\int_0^1|g_{\log}(1-|x-y|)|\,dy.
\]
It is symmetric about $1/2$.  For $0\le x\le1/2$,
\begin{equation}
M_{\log}(x)
=\int_x^1|g_{\log}(r)|\,dr
 +\int_{1-x}^1|g_{\log}(r)|\,dr.
\label{eq:glog-row-sum}
\end{equation}
Put $k(x)=g_{\log}(1-x)-g_{\log}(x)$.  The last inequality in
\eqref{eq:glog-root-certificate} gives $k'(x)<0$ and $k(1/2)=0$, so
$M_{\log}'(x)>0$ on $[0,\alpha]$ and $M_{\log}'(x)<0$ on
$[1-\beta,1/2]$.  On $[\alpha,1-\beta]$ one has
$M_{\log}'(x)=h(x):=g_{\log}(x)+g_{\log}(1-x)$.  Strict convexity makes $h$
strictly decreasing there, and the same rational enclosure gives
\[
h(0.3144)>0>h(0.3145).
\]
Thus the unique maximizer is $x_*\in(0.3144,0.3145)$.  Integration of
\eqref{eq:glog-row-sum} by the antiderivative $G_{\log}'$ gives
\[
M_{\log}(x_*)
=2G_{\log}'(1)-G_{\log}'(1-x_*)
 +G_{\log}'(x_*)-2G_{\log}'(\beta).
\]
Applying the same rational Taylor enclosure on the certified intervals for
$\beta$ and $x_*$ yields the deliberately rounded bound
\begin{equation}
\boxed{\sup_{0\le x\le1}M_{\log}(x)<4.6247.}
\label{eq:Glog-Schur}
\end{equation}
All inequalities in
\eqref{eq:glog-root-certificate}--\eqref{eq:Glog-Schur} are interval
consequences of \eqref{eq:exp-rational-bound}; no floating-point value enters
the proof.

Schur's test and \eqref{eq:primitive-identity} now give
\[
\langle K_{G_{\log}}q,q\rangle
>(4.6770-4.6247)\|Q\|_2^2
=0.0523\|Q\|_2^2
>\kappa_*Q_\triangle(q).
\]
The packet inequality follows by congruence through the stable-core analysis
map used in the definition of $P_T$.  This proves the proposition.
\end{proof}

\begin{hypothesis}[One-$\chi$ one-sided relative spectral defect]
\label{hyp:one-chi-relative-defect}
For the same packet centres selected by Lemma~\ref{lem:hermite-packet-evaluation}
and the associated retained space $\Pret$ used in
Section~\ref{sec:multiplicity-endgame}, there is a deterministic sequence
$\lambda_T\ge\lambda_*>0$ such that, with
\begin{equation}
\mathcal D_T
:=\Gfr_T^{-1/2}(A_T^{\rm true}-\lambda_TP_T)\Gfr_T^{-1/2},
\label{eq:true-model-relative-defect}
\end{equation}
one has
\begin{equation}
\boxed{
\beta_-(T;\lambda_T)
:=\frac{\tr[(\mathcal D_T)_-^2]}
{(\kappa_*\lambda_T)^2N_T}
=O(L^{-2}).}
\label{eq:one-chi-relative-defect}
\end{equation}
Only the negative part is constrained.  In particular, no two-sided operator
norm or Hilbert--Schmidt approximation of $A_T^{\rm true}$ by
$\lambda_TP_T$ is assumed.
\end{hypothesis}

\begin{remark}[Unresolved input]
Hypothesis~\ref{hyp:one-chi-relative-defect} is not proved in this manuscript.
The surviving phase $e(-ne^d)$ is precisely why no scalar reduction is used to
establish it.  The normalization is chosen so that
$\beta_-(T;\lambda_T)=o(1)$ deletes only $o(N_T)$ spectral directions; the
displayed $O(L^{-2})$ rate gives the quantitative conditional conclusion in
Theorem~\ref{thm:main}.
\end{remark}

\begin{remark}[Finite certificate]
The inequalities above are finite rational interval calculations obtained from
\eqref{eq:exp-rational-bound} on the displayed mesh.  No asymptotic numerical
input is used in the floor estimate.
\end{remark}

\begin{lemma}[The whole-space floor survives restriction]
\label{lem:model-compression}
Let $V$ be any subspace of the finite packet coordinate space and let $P_V$ be
its orthogonal projection. Then
\[
P_VP_TP_V\succeq\kappa_*\,P_V\Gfr_TP_V
\quad\text{on }V.
\]
\end{lemma}

\begin{proof}
For $v\in V$, $P_Vv=v$ and Proposition~\ref{prop:model-floor} applies directly.
\end{proof}

For a Hermitian lag form $B$ on the mean-zero fibre define
\begin{equation}
\|B\|_{\op,\triangle}
:=\sup_{\substack{q\ne0\\ \int q=0}}
\frac{|\langle Bq,q\rangle|}{Q_\triangle(q)}.
\label{eq:triangular-form-norm}
\end{equation}

\begin{proposition}[Frozen principal floor]
\label{prop:finiteT-principal-comparison}
On $V_{\rm mean}$,
\begin{equation}
\boxed{P_T\succeq\kappa_*\,\Gfr_T.}
\label{eq:principal-floor}
\end{equation}
This is the sole numerical floor used downstream.
\end{proposition}

\begin{proof}
This is Proposition~\ref{prop:model-floor} followed by restriction to
$V_{\rm mean}$.
\end{proof}
