\subsection{Revised log-only remainder ledger}
\label{sec:logonly-remainder-ledger}

This subsection replaces the remainder assignments that depended on the
one-sided packet--HLZ bridge for the principal logarithmic faces.  The abstract
primitive criterion of Lemma~\ref{lem:two-variable-primitive}, the shell
normalization, the nonstationary estimates, and the genuinely geometric edge
families remain unchanged.

\begin{lemma}[Lagwise HLZ contour error is relative-regular]
\label{lem:lagwise-HLZ-error-regular}
Let $R_{\log,T}^{\rm HLZ}$ be the error in
Proposition~\ref{prop:lagwise-packet-HLZ} after the finite Cauchy residues and
curvature extraction.  On every retained compression of dimension $O(N_T)$,
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{\log,T}^{\rm HLZ}\Gfr_T^{-1/2}\right\|_{S_2}^2
=o(N_T).
}
\label{eq:lagwise-HLZ-relative-HS}
\end{equation}
\end{lemma}

\begin{proof}
Let $\varepsilon_T(\rho)$ denote any one of the finitely many scalar error
profiles produced after the separated Cauchy residues.  By
Lemma~\ref{lem:packet-HLZ-diagonal} and
Proposition~\ref{prop:lagwise-packet-HLZ}, for every prescribed $A>0$,
\begin{equation}
\max_{0\le j\le2}\sup_\rho
|\partial_\rho^j\varepsilon_T(\rho)|
\ll L^{-A+C_0}
\label{eq:lagwise-HLZ-profile-saving}
\end{equation}
for one fixed loss $C_0$ after all fixed derivatives have been included.

On a unit physical shell $x=J_j+t$, $y=J_j+s$, the corresponding normalized
kernel is a finite sum of terms
\begin{equation}
D_{j,T}^{\rm HLZ}(t,s)
=\frac1L a_{j,T}(t,s)
\varepsilon_T\!\left(
\nu_{j,T}(t,s)-
\frac{|\log(J_j+t)-\log(J_j+s)|}{L}
\right),
\label{eq:lagwise-HLZ-shell-kernel}
\end{equation}
where $|\nu_{j,T}-1|\ll L^{-1}$ and the smooth amplitude satisfies the first
two derivative bounds of Lemma~\ref{lem:shell-normalization-transfer}.  These
bounds follow from Lemma~\ref{lem:stationary-symbol} and the bounded-overlap
height partition.

Apply Lemma~\ref{lem:shell-normalization-transfer} with
$H=\varepsilon_T$.  After division by the exact primitive reference density,
the diagonal jump is
\[
\sup_{j,t}\|\mathcal J_{j,T}^{\rm HLZ}(t)\|_{\op}
\ll L^{-A+C_1}=o(1),
\]
while the normalized regular mixed derivative has per-shell squared
Hilbert--Schmidt mass
\[
\iint\|\mathcal K_{j,T}^{\rm HLZ}(t,s)\|_{S_2}^2\,dt\,ds
\ll L^{-2A+C_2}.
\]
The number of retained shells is $O(X_T)=O(TL^{-A_0})$.  Choose the arbitrary
HLZ saving exponent $A$ so large that
$-A_0-2A+C_2<-2$.  Then the shell sum is $o(TL)=o(N_T)$.  The two-variable
primitive criterion \eqref{eq:primitive-HS-criterion} proves
\eqref{eq:lagwise-HLZ-relative-HS}.
\end{proof}

\begin{lemma}[The exact direct residual is relative-regular]
\label{lem:deltaH-dir-relative-regular}
Let $R_{\rm dir,T}$ denote the right-edge contribution of
$\mathcal C\Delta H_{\rm dir}$ in \eqref{eq:logonly-product-split}, including
its adjoint orientation and the fixed stationary, height-block, and curvature
normalizations.  Then
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{\rm dir,T}\Gfr_T^{-1/2}\right\|_{S_2}^2
=o(N_T).
}
\label{eq:deltaH-dir-relative-HS}
\end{equation}
No low-rank compressed-shift decomposition is used.
\end{lemma}

\begin{proof}
Lemma~\ref{lem:deltaH-dir-small} gives, on the safe line and through all fixed
source derivatives,
\[
\Delta H_{\rm dir}=O(L^{-2})
\]
up to a fixed polylogarithmic factor.  Keeping the exact stationary lag
geometry, the shell-local normalized kernel is therefore a finite sum of terms
\begin{equation}
D_{j,T}^{\rm dir}(t,s)
=\frac1L a_{j,T}(t,s)
H_{j,T}^{\rm dir}\!\left(
\nu_{j,T}(t,s)-
\frac{|\log(J_j+t)-\log(J_j+s)|}{L}
\right),
\label{eq:deltaH-dir-shell-kernel}
\end{equation}
with
\begin{equation}
\max_{0\le r\le2}\sup_\rho
|\partial_\rho^rH_{j,T}^{\rm dir}(\rho)|
\ll L^{-2+C_{\rm dir,reg}}
\label{eq:deltaH-dir-profile-bound}
\end{equation}
for one fixed exponent $C_{\rm dir,reg}$ before the global choice of $A_0$.
The smooth amplitude $a_{j,T}$ satisfies the derivative bounds of
Lemma~\ref{lem:stationary-symbol}; no Fourier sideband expansion is needed.

Lemma~\ref{lem:shell-normalization-transfer} and
Lemma~\ref{lem:two-variable-primitive} then give a normalized diagonal jump
$O(L^{-2+C})$ and a fixed-polylogarithmic per-shell mixed-derivative bound.
Choose $A_0$ after this fixed exponent, exactly as in
\eqref{eq:acyclic-A0-choice}, so that the shell sum is $o(N_T)$.  This proves
\eqref{eq:deltaH-dir-relative-HS}.
\end{proof}

\begin{lemma}[Log-only $f'$ class is relative-regular]
\label{lem:logonly-fprime-relative-regular}
Let $R_{f',T}^{\log}$ be the contribution of
\eqref{eq:logonly-fprime-source}.  Then
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{f',T}^{\log}\Gfr_T^{-1/2}\right\|_{S_2}^2
=o(N_T).
}
\label{eq:logonly-fprime-relative-HS}
\end{equation}
\end{lemma}

\begin{proof}
Every term in \eqref{eq:logonly-fprime-source} contains $f'=O(T^{-1})$ on the
retained height strip.  The reciprocal $(f-P)^{-1}$ is $O(L^{-1})$ on the safe
line.  The fixed stationary, shell, curvature, and Cauchy derivatives cost only
a fixed power of $L$.  Thus the existing $f'$ proof in
Proposition~\ref{prop:regular-ledger} applies verbatim, with a strictly smaller
source family.
\end{proof}

\begin{proposition}[Revised principal plus remainder decomposition]
\label{prop:logonly-final-decomposition}
On the retained space $\Pret$, the contour matrix has the decomposition
\begin{equation}
\boxed{
A_T=P_T+E_T^{\rm geom}+R_T^{\log},
}
\label{eq:logonly-final-decomposition}
\end{equation}
where $P_T=A_T^{G_{\log}}$ is defined by
\eqref{eq:PT-logonly-definition},
\begin{equation}
\rank E_T^{\rm geom}=o(N_T),
\label{eq:logonly-edge-rank}
\end{equation}
and
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_T^{\log}\Gfr_T^{-1/2}\right\|_{S_2}^2=o(N_T).
}
\label{eq:logonly-total-relative-HS}
\end{equation}
The operator $E_T^{\rm geom}$ contains only the genuinely geometric stationary
buffer, exterior packet support, horizontal-collar, and other source-independent
finite-rank families already proved in Sections~\ref{sec:shell-frame}--\ref{sec:edge-rank}.
It does not contain the principal logarithmic source $H_{\log}$.
\end{proposition}

\begin{proof}
By Lemma~\ref{lem:holomorphic-self-elimination}, the critical-$J$ self block is
removed before principal reduction.  Proposition~\ref{prop:lagwise-packet-HLZ}
and Corollary~\ref{cor:frozen-logonly-lag-kernel} identify the unique order-one
principal source with $P_T=A_T^{G_{\log}}$.  The finite-height parameter defect
is relative-regular by the same proof as
Lemma~\ref{lem:finiteT-parameter-regular}; its scalar profile is now simply the
two-face version of the old defect.  The scalar HLZ contour error is covered by
Lemma~\ref{lem:lagwise-HLZ-error-regular}, the exact direct residual by
Lemma~\ref{lem:deltaH-dir-relative-regular}, and the $f'$ terms by
Lemma~\ref{lem:logonly-fprime-relative-regular}.  Higher stationary/Stirling
terms and nonstationary leakage retain the existing estimates.

The critical-$J$ AFE smooth-to-sharp class, the frozen-self class, and the
low/high sideband decomposition of the logarithmic principal source are absent
from the revised source ledger.  The remaining finite-rank terms are precisely
the already established geometric edge families, whose total rank is
$o(N_T)$.  Summing the relative-Hilbert--Schmidt classes proves
\eqref{eq:logonly-total-relative-HS}.
\end{proof}

\begin{remark}[Section 11 direct-shift machinery]
The compressed-shift estimates of Section~\ref{sec:edge-rank} remain available
for genuinely one-sided auxiliary or exterior-support terms for which the
physical variable is a one-sided coordinate.  They are not used to represent
the packet-polarized logarithmic principal source, whose exact variable is the
lag $d=y-x$ by Proposition~\ref{prop:lagwise-packet-HLZ}.
\end{remark}
