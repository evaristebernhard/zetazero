\section{Lagwise packet--HLZ bridge for the log-only source}
\label{sec:lagwise-packet-hlz}

For the principal logarithmic source we keep the packet polarization in its
actual continuous lag variable $d=y-x$.  No Fourier multiplier of a lag
convolution is identified with a one-sided packet-prefix multiplier.

Fix the stable logarithmic core
\begin{equation}
\boxed{
W_T:=L-A_0\log L,
\qquad
X_T:=e^{W_T}=\frac{T}{2\pi L^{A_0}}.
}
\label{eq:stable-log-core-main}
\end{equation}
On the translated flat plateau define the packet synthesis
\begin{equation}
\boxed{
(\mathcal A_Tv)(x):=\phi_L(x)
\sum_{\nu\in\mathcal I_T}v_\nu e^{-i\tau_\nu x}.
}
\label{eq:lagwise-packet-analysis}
\end{equation}
Only its restriction to pairs with $|x-y|\le W_T$ enters the principal core.

\begin{proposition}[Lagwise packet--HLZ bridge]
\label{prop:lagwise-packet-HLZ}
Restrict to the invariant source $h=k=1$, one retained height block, and the
stable flat packet core. Put
\[
X(\tau)=\frac{\tau}{2\pi},\qquad
\lambda(\tau)=\frac{\log X(\tau)}L,
\]
and, for $|d|\le W_T$, define
\begin{equation}
\boxed{\rho_\tau(d):=\lambda(\tau)-\frac{|d|}{L}.}
\label{eq:lagwise-rho}
\end{equation}
For the logarithmic source $H_{\log}$ of \eqref{eq:Hlog}, the principal
stationary packet form after the finite three-$Z$ Cauchy residues is
\begin{align}
\mathcal Q_{\log,T}(v,w;p)
={}&\iint
 (\mathcal A_Tv)(x)\overline{(\mathcal A_Tw)(y)}
 \mathcal G^{\log}_{\lambda(\tau),\rho_\tau(y-x)}(p)\,dx\,dy
\notag\\
&+\mathcal R^{\rm HLZ}_{\log,T}(v,w;p),
\label{eq:lagwise-HLZ-form}
\end{align}
where
\begin{equation}
\boxed{
\mathcal G^{\log}_{\lambda,\rho}(p)
=e^{\lambda p}\left[
-2E_\rho(p)^2
+e^{2\lambda}E_\rho(p+2)E_\rho(p-2)
\right],
\qquad
E_\rho(q)=\int_0^\rho e^{-qt}\,dt.
}
\label{eq:logonly-local-profile}
\end{equation}
For every prescribed $A>0$, after the fixed Cauchy and curvature residues have
been taken, the scalar error profile and its first two $\rho$-derivatives are
$O_A(L^{-A})$ up to a fixed polylogarithmic factor, uniformly in
$|d|\le W_T$ and in the retained height blocks.
\end{proposition}

\begin{proof}
For basis packets $e_r,e_s$, exact polarization gives
\begin{equation}
\mathcal B_{e_r,e_s}(c+it)
=\iint\phi_L(x)\phi_L(y)e^{-i\tau_sx+i\tau_ry}
 e^{\sigma_0(x-y)}e^{it(x-y)}\,dx\,dy,
\label{eq:lagwise-polarization-recall}
\end{equation}
with $\sigma_0=c-1/2$. For one Dirichlet monomial $\xi^{-c-it}$ put
$d=y-x$. Then exactly
\[
e^{\sigma_0(x-y)}e^{it(x-y)}\xi^{-c-it}
=e^{d/2}(\xi e^d)^{-c-it}.
\]
Hence the stationary dilation parameter of Section~\ref{sec:stationary} is
\begin{equation}
\boxed{w=d=y-x.}
\label{eq:true-packet-lag}
\end{equation}
It is a continuous lag, not a displacement attached to either packet label.
The principal stationary calculation therefore depends on the two packet legs
only through $d$.

By Lemma~\ref{lem:scalar-HLZ-local-height}, the scalar diagonal Mellin factor
may be written at local scale as
\[
\left(\frac{X(\tau)^2}{\ell^2}\right)^u
\]
without changing its $u=0$ residue or its arbitrary-log contour error.  Under
the exact stationary dilation, the two Hermitian orientations have cutoffs
\[
\ell\le X(\tau)e^{-d},\qquad
\ell\le X(\tau)e^{d}.
\]
Their intersection is therefore
\begin{equation}
\boxed{\ell\le X(\tau)e^{-|d|}.}
\label{eq:lagwise-common-cutoff}
\end{equation}
Writing $t=\log\ell/L$ gives
$0\le t\le\rho_\tau(d)$.  By \eqref{eq:stable-log-core-main},
$|d|\le W_T$ makes this endpoint positive for all sufficiently large $T$ and
keeps it in a fixed compact interval.

Fix $d,p$ and the separated Cauchy variable $\mathbf z$.  After the universal
stationary carrier and the two packet legs have been removed, the remaining
scalar Mellin weight has the form
\begin{equation}
W_d(u,\mathbf z,p)
=\int_0^{\rho_\tau(d)}
 a_{d,\mathbf z,p,T}(t)e^{Lut}\,dt,
\label{eq:lagwise-Mellin-weight}
\end{equation}
or a fixed finite product of such factors.  Lemma~\ref{lem:stationary-symbol}
gives fixed-order derivative bounds for the stationary amplitude by a fixed
power of $L$; the separated shifts $|z_j|\asymp L^{-1}$ cost only another fixed
polylogarithmic factor.  Moving the endpoint in
\eqref{eq:lagwise-Mellin-weight} creates only finitely many endpoint jets.
Consequently \eqref{eq:lagwise-Mellin-weight} satisfies
\eqref{eq:scalar-HLZ-weight-bounds} uniformly for $|d|\le W_T$.

Lemma~\ref{lem:scalar-lagweight-HLZ} can therefore be applied pointwise in the
actual lag $d$.  Its arbitrary-log error remains uniform through the first two
endpoint and curvature derivatives.  Proposition~\ref{prop:scalar-HLZ-three-face}
then gives the finite three-face residues.  For the source $-2P$, the two
one-sided diagonal factors are both $E_\rho(p)$, hence
\[
-2e^{\lambda p}E_\rho(p)^2.
\]
For $P(s-2/L)$, the translated exponents are $p+2$ and $p-2$, while the
functional-equation height tag is
$X^{2/L}=e^{2\lambda}$; this gives
\[
e^{\lambda p}e^{2\lambda}E_\rho(p+2)E_\rho(p-2).
\]
Their sum is \eqref{eq:logonly-local-profile}.

On the original absolutely convergent scalar contour the $x,y$ variables lie
in compact packet support and the $u$-integral has Gaussian vertical decay.
Thus Fubini is legitimate before the scalar contour is moved.  After the move,
the uniform scalar error can be integrated against the same two packet legs.
This proves \eqref{eq:lagwise-HLZ-form} and the differentiated error estimate.
\end{proof}

\begin{corollary}[Frozen log-only lag kernel]
\label{cor:frozen-logonly-lag-kernel}
Let
\begin{equation}
\boxed{
G_{\log}(\rho):=[p^2]\,
 e^p\left[-2E_\rho(p)^2+e^2E_\rho(p+2)E_\rho(p-2)\right].
}
\label{eq:Glog-from-source}
\end{equation}
After freezing $\lambda(\tau)=1$ and rescaling $r=x/L$, $s=y/L$, the principal
operator is the pullback of the genuine lag kernel
\begin{equation}
\boxed{K_{\log}(r,s)=G_{\log}(1-|r-s|).}
\label{eq:Klog-lag-kernel}
\end{equation}
The difference between the finite-height profile
$G_{\log,\lambda(\tau)}(\lambda(\tau)-|x-y|/L)$ and the frozen kernel has its
first two normalized lag derivatives $O(L^{-1})$ and belongs to the
finite-parameter regular class.
\end{corollary}

\begin{proof}
The map $(\lambda,\rho,p)\mapsto\mathcal G^{\log}_{\lambda,\rho}(p)$ is analytic
on a fixed neighbourhood of the retained parameter set and
$\lambda(\tau)=1+O(L^{-1})$.  The mean-value theorem gives the required
$O(L^{-1})$ estimates through two $\rho$-derivatives.  Substitution of
$d=y-x$ in Proposition~\ref{prop:lagwise-packet-HLZ} gives the lag kernel.
\end{proof}

\begin{remark}[No one-sided principal bridge]
The proof above ends with the lag kernel itself.  Fourier diagonalization may
of course be used as an abstract unitary representation of a convolution, but
no subsequent step identifies its multiplier with pointwise modulation in a
one-sided packet coordinate.  In particular, the compressed-shift machinery is
not part of the logarithmic principal reduction.
\end{remark}
