\subsection{Log-only revision of the right-edge principal source}
\label{sec:logonly-source-revision}

This subsection replaces the use of the frozen archimedean self term in the
principal model.  The matrix-adjoint reflection identities above remain
unchanged.  What changes is the source ledger after the two vertical sides have
been reassembled.

\begin{lemma}[Holomorphic elimination of the archimedean self source]
\label{lem:holomorphic-self-elimination}
For every fixed normalized curvature derivative $0\le j\le2$,
\begin{equation}
\boxed{
\partial_p^j\oint_{\partial\mathcal R_T^{\rm sym}}
 f(s)C_{\mathscr Z}^{(p)}(s)\mathcal B_{v,w}(s)\,ds=0.
}
\label{eq:holomorphic-self-elimination}
\end{equation}
Consequently the complete $f$-self block may be removed before any AFE,
stationary, packet, or HLZ reduction.  Its vertical contribution is exactly the
negative of its horizontal contribution, and on the retained packet space the
latter is covered by Lemma~\ref{lem:horizontal-decay}.
\end{lemma}

\begin{proof}
The three factors $f$, $C_{\mathscr Z}^{(p)}$, and $\mathcal B_{v,w}$ are
holomorphic in the high rectangle.  The fixed $p$-derivatives remain
holomorphic in the same rectangle, so Cauchy's theorem gives
\eqref{eq:holomorphic-self-elimination}.  Splitting the closed contour into its
vertical and horizontal pieces gives the second assertion.  No Dirichlet
expansion or approximate functional equation is involved.
\end{proof}

Retain the sign convention \eqref{eq:PQ-convention} and put
\begin{equation}
\boxed{
H_{\log}(s):=-2P(s)+P(s-2/L),
}
\label{eq:Hlog}
\end{equation}
\begin{equation}
\boxed{
\Delta H_{\rm dir}(s)
:=P(s)+\frac{Q(s)}{f(s)-P(s)}-P(s-2/L).
}
\label{eq:deltaH-dir}
\end{equation}
Since the exact right-edge logarithmic derivative is
\[
L_1(s)=-P(s)+\frac{f'(s)+Q(s)}{f(s)-P(s)},
\]
we have the exact identity
\begin{equation}
\boxed{
L_1(s)=H_{\log}(s)+\Delta H_{\rm dir}(s)
      +\frac{f'(s)}{f(s)-P(s)}.
}
\label{eq:L1-logonly-split}
\end{equation}
Thus the principal arithmetic source contains no $f$-self term and no frozen
quantity $F_b$.

\begin{lemma}[Product-level $f'$ separation for the log-only route]
\label{lem:logonly-fprime-separation}
Let $\mathcal C=\zeta^2Q$ as in Lemma~\ref{lem:curvature-source}.  Then the
right-edge product after holomorphic self elimination is exactly
\begin{equation}
\boxed{
q^2\left[
 \mathcal C H_{\log}
 +\mathcal C\Delta H_{\rm dir}
 +R_{f'}^{\log}
\right],
}
\label{eq:logonly-product-split}
\end{equation}
where
\begin{equation}
\boxed{
R_{f'}^{\log}
=\frac{f'}{f-P}\mathcal C
 +f'\zeta^2H_{\log}
 +f'\zeta^2\Delta H_{\rm dir}
 +\frac{(f')^2}{f-P}\zeta^2.
}
\label{eq:logonly-fprime-source}
\end{equation}
Every summand in $R_{f'}^{\log}$ contains at least one factor $f'$ and hence
belongs to the regular $f'$ class.
\end{lemma}

\begin{proof}
Use
$C_{\mathscr Z}/q^2=\mathcal C+f'\zeta^2$ together with
\eqref{eq:L1-logonly-split} and multiply out.  This gives
\eqref{eq:logonly-product-split}--\eqref{eq:logonly-fprime-source} without an
approximation.  On fixed strips $f'=O(T^{-1})$, while $f-P\asymp L$ on the
safe right edge, so every displayed $f'$ term has the regular gain required by
the later shell estimates.
\end{proof}

\begin{lemma}[Second-order smallness of the direct residual]
\label{lem:deltaH-dir-small}
Fix the absolute-convergence line $\Re s=1+\delta_0>1$.  Uniformly for
$t\asymp T$ and every fixed derivative order used in the source ledger,
\begin{equation}
\boxed{
\partial_s^j\Delta H_{\rm dir}(s)=O_{\delta_0,j}(L^{-2}),
\qquad j\ \text{fixed}.
}
\label{eq:deltaH-dir-small}
\end{equation}
The same conclusion holds after the fixed normalized curvature derivatives,
up to a fixed polylogarithmic factor.
\end{lemma}

\begin{proof}
On the fixed safe line, $P,Q$ and every fixed derivative are $O_{\delta_0}(1)$,
while $f=L/2+O_{\delta_0}(1)$.  Hence
\[
\frac1{f-P}=\frac2L+O(L^{-2}).
\]
With $\delta=2/L$ and $P'=-Q$, Taylor's theorem gives
\[
P(s-\delta)=P(s)-\delta P'(s)+O(\delta^2)
           =P(s)+\frac2LQ(s)+O(L^{-2}).
\]
Also
\[
\frac{Q(s)}{f(s)-P(s)}=\frac2LQ(s)+O(L^{-2}).
\]
Substitution into \eqref{eq:deltaH-dir} proves the case $j=0$.
Differentiating a fixed number of times uses the same safe-line bounds and the
fact that every derivative of $(f-P)^{-1}$ is $O(L^{-2})$ or smaller after the
leading $2/L$ constant has disappeared.  The normalized curvature derivatives
only insert fixed powers of $\log n/L$ and the already controlled height
symbols.
\end{proof}

For the revised principal route the only order-one right-edge source retained
for scalar HLZ is therefore $\mathcal C H_{\log}$.  The critical-$J$ AFE and
the frozen self term are not part of the principal model; the two regular
families are $\mathcal C\Delta H_{\rm dir}$ and
$R_{f'}^{\log}$.
