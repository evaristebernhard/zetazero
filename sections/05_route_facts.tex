\section{Invariant-source facts used across the reduced route}
\label{sec:route-facts}

\begin{proposition}[Exterior packet support is geometric]
\label{prop:exterior-packet-support}
The contribution for which at least one packet leg lies outside the stable
physical core is included in the geometric decomposition of
Proposition~\ref{prop:geometric-edge-ledger}.  Its stationary part has rank
$o(N_T)$ and its complementary nonstationary part is
relative-Hilbert--Schmidt $o(N_T)$ with respect to $\Gfr_T$.
\end{proposition}

\begin{proof}
This is precisely the exterior packet-support component in the definition of
$E_T^{\rm geom}$ and $R_T^{\rm geom}$ in
Proposition~\ref{prop:geometric-edge-ledger}.  The proof there uses only
stationary row counting, the log-only coefficient square bound, and repeated
nonstationary integration by parts.
\end{proof}
