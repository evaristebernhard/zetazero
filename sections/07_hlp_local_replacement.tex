\section{Exact HLP levels and the smooth packet local replacement}
\label{sec:hlp-local}

Fix once and for all a spectral right edge
\[
\Re s=1+\delta,\qquad \delta>0.
\]
All Dirichlet expansions in this section are first made on this line. Absolute
convergence gives $P(s)=O_\delta(1)$ and $Q(s)=O_\delta(1)$, while
$f(s)=\frac12L+O_\delta(1)$ on $t\asymp T$. Hence
\[
\left|\frac{P(s)}{f(s)}\right|\ll_\delta L^{-1},
\]
The coefficient organization is the one appearing in the local-extrema analysis
of Hughes--Lugmayer--Pearce-Crump~\cite{HLP}, but no RH-dependent estimate from
that paper is used here.  Indeed, on the present fixed safe line the exact
identity
\begin{equation}
\boxed{
\frac{Q(s)}{f(s)-P(s)}
=\sum_{k\ge1}f(s)^{-k}P(s)^{k-1}Q(s)
}
\label{eq:safe-line-HLP-expansion}
\end{equation}
follows directly from the geometric series, and converges absolutely and
uniformly because $|P/f|\ll_\delta L^{-1}$.  Thus the levelwise expansion used
below is unconditional before any contour movement.  It is sometimes useful to
keep the corresponding finite algebraic identity separate.  For every
$K\ge1$,
\begin{equation}
\boxed{
\frac{Q}{f-P}
=\sum_{k=1}^{K}f^{-k}P^{k-1}Q
+\frac{P^KQ}{f^K(f-P)}.
}
\label{eq:finite-HLP-resolvent}
\end{equation}
Equivalently, if $|P/f|\le\rho<1$, the single exact tail satisfies
\[
\left|\frac{P^KQ}{f^K(f-P)}\right|
\le \left|\frac Qf\right|\frac{\rho^K}{1-\rho}.
\]
This finite resolvent identity is only an algebraic bookkeeping device.  In the
carrier decomposition below the unprojected high exact levels are deliberately
kept levelwise on the safe right edge, because their physical $M$-carrier must
remain visible; they are not silently replaced by the scalar resolvent tail.
Since $f'(s)=O(T^{-1})$, its separated $f'$ contribution is also
$O_\delta((T L)^{-1})$ on this line.
The exact multiplier at level $k$ changes only the Dirichlet coefficient
sequence. If the total convolution index is $M$, the stationary phase remains
\begin{equation}
t_*=2\pi My,
\qquad
\text{carrier}=e(-My).
\label{eq:total-carrier}
\end{equation}
Thus exact levels do not introduce a new moving support variable. Their packet
weights remain in the finite-interval Laplace/Mellin class
\eqref{eq:joint-weight}.

\subsection{Uniform factorial control of the HLP hierarchy}
For $k\ge0$ define the nonnegative Dirichlet-convolution coefficients
$\Lambda_k$ by
\[
P(s)^k=\sum_{n\ge1}\frac{\Lambda_k(n)}{n^s},
\qquad \Lambda_0(1)=1,
\]
and for $k\ge1$ let $\alpha_k$ be the coefficient sequence of
$P^{k-1}Q$.

\begin{lemma}[Derivative identity and support]
\label{lem:alpha-identity}
For every $k\ge1$,
\begin{equation}
\boxed{
\alpha_k(n)=\frac{\log n}{k}\Lambda_k(n),
\qquad
\Lambda_k(n)\ne0\Longrightarrow n\ge2^k.
}
\label{eq:alpha-identity}
\end{equation}
\end{lemma}

\begin{proof}
Differentiate the identity for $P^k$. Since $P'=-Q$,
\[
-\sum_n(\log n)\Lambda_k(n)n^{-s}
=kP^{k-1}P'
=-kP^{k-1}Q
=-k\sum_n\alpha_k(n)n^{-s}.
\]
Coefficient comparison gives the first identity.  Equivalently, this is a
pure coefficient-algebra statement: multiplication by $\log n$ is a derivation
for Dirichlet convolution, so the same identity may be proved without analytic
termwise differentiation once the convolution coefficients are defined.  Every
nonzero term of the $k$-fold convolution defining $\Lambda_k$ is a product of
integers at least $2$, giving the support statement.
\end{proof}

\begin{proposition}[Uniform factorial square-sum majorant]
\label{prop:factorial-majorant}
There are absolute constants $A,C>0$ such that, for every $x\ge2$ and
$k\ge1$,
\begin{equation}
\boxed{
\sum_{n\le x}\alpha_k(n)^2
\le C\frac{A^k}{k!}\,
 x\bigl(\log(ex)\bigr)^{2k+1}.
}
\label{eq:factorial-majorant}
\end{equation}
\end{proposition}

\begin{proof}
Put
\[
S_k(x):=\sum_{n\le x}\Lambda_k(n)^2.
\]
The convolution identity
\[
\Lambda_k(n)=\sum_{d\mid n}\Lambda(d)\Lambda_{k-1}(n/d)
\]
and $\sum_{d\mid n}\Lambda(d)=\log n$ give by weighted Cauchy--Schwarz
\[
\Lambda_k(n)^2
\le(\log n)\sum_{d\mid n}\Lambda(d)\Lambda_{k-1}(n/d)^2.
\]
Hence
\begin{equation}
S_k(x)\le(\log x)\sum_{d\le x}\Lambda(d)S_{k-1}(x/d).
\label{eq:Sk-recurrence}
\end{equation}
Chebyshev's estimate $\psi(y)=\sum_{n\le y}\Lambda(n)\ll y$ and Stieltjes
integration by parts imply, uniformly for $0\le v\ll\log(ex)$,
\begin{equation}
\sum_{d\le x}\frac{\Lambda(d)}d
\left(\log\frac{ex}{d}\right)^v
\ll\frac{(\log(ex))^{v+1}}{v+1}.
\label{eq:weighted-Chebyshev}
\end{equation}
Indeed, apply partial summation to
$d^{-1}(\log(ex/d))^v$ and use $\psi(t)\ll t$.

Inducting in \eqref{eq:Sk-recurrence}, starting from
$S_1(x)\ll x\log(ex)$, and applying
\eqref{eq:weighted-Chebyshev} with $v=2k-3$ gives
\[
S_k(x)
\le C\frac{A^{k-1}}{(k-1)!}
 x(\log(ex))^{2k-1}.
\]
If $S_k(x)\ne0$, Lemma~\ref{lem:alpha-identity} gives $2^k\le x$. Since
$\log2>1/2$, this implies $k\le2\log x$, and for the value $v=2k-3$ used above
one has the explicit bound
\[
v+1=2k-2\le4\log(ex).
\]
Thus the uniformity condition in \eqref{eq:weighted-Chebyshev} may be taken
with one fixed constant; there is no hidden fixed-$k$ restriction in the
induction. Finally
\[
\sum_{n\le x}\alpha_k(n)^2
\le\frac{(\log x)^2}{k^2}S_k(x),
\]
and enlarging $A,C$ proves \eqref{eq:factorial-majorant}.
\end{proof}

\begin{lemma}[Quadratic bound for direct physical carriers]
\label{lem:direct-carrier-square}
Fix $C_x\ge0$. On the fixed absolute-convergence right edge, after the common
shifted-curvature factor and the deterministic stationary/shell normalization
defining $\mathcal K_{j,T}$ have been applied, let $c_m^{\rm dir}(p)$ denote the
remaining normalized coefficient of a direct one-$\chi$ carrier. For the
ordinary direct logarithmic-derivative sources, the exact HLP levels on this
safe-line representation, and their fixed coefficient shifts, uniformly for
$x\le T L^{C_x}$ and $0\le q\le2$,
\begin{equation}
\boxed{
\sum_{m\le x}
\left|\partial_p^q c_m^{\rm dir}(p)\right|^2
\ll_{C_x} xL^{C_{\rm dir}}.
}
\label{eq:direct-carrier-square}
\end{equation}
Here $C_{\rm dir}$ may depend on the fixed value of $C_x$. The same bound holds
for the full norm-summable HLP hierarchy, for any subset of its levels, and for
every support restriction of the direct carrier family.
\end{lemma}

\begin{proof}
By Proposition~\ref{prop:trivial-outer-twist}, no nontrivial outer arithmetic
map remains.  On the fixed absolute-convergence right edge every exact HLP
level is a genuine direct source before any optional local projection is
introduced. For $k\ge1$ its changing internal multiplier is exactly
$f^{-k}P^{k-1}Q$, with coefficient $f^{-k}\alpha_k(m)$. Since $C_x$ is fixed and $x\le T L^{C_x}$,
\[
\log(ex)=L+O_{C_x}(\log L)\asymp_{C_x}L,
\]
while $|f|\asymp L$ on the fixed safe line. Hence
\eqref{eq:factorial-majorant} gives
\[
\left(\sum_{m\le x}|f^{-k}\alpha_k(m)|^2\right)^{1/2}
\ll (xL)^{1/2}\frac{C^k}{\sqrt{k!}}.
\]
This is the deterministic normalization cancellation behind the direct-carrier
bound: the factor $L^{-k}$ absorbs the full $2k$-fold logarithmic growth in the
square-sum majorant, leaving only the single factor $L^{1/2}$.  A fixed $p$-derivative introduces only a polynomial in $k$ and factors
\[
\frac{\log m}{L}\le \frac{\log(TL^{C_x})}{L}=1+O_{C_x}\!\left(\frac{\log L}{L}\right)=O_{C_x}(1).
\]
Minkowski's inequality then reduces the level assembly to
\[
\sum_{k\ge1}\frac{C^k(1+k)^q}{\sqrt{k!}}<\infty
\qquad(q\text{ fixed}).
\]
For completeness, this summability is structural rather than a delicate ratio
test: $k+1\le2^k$ absorbs the polynomial factor into the exponential parameter,
and
\[
\frac{A^k}{\sqrt{k!}}
=\frac{(2A)^k}{\sqrt{k!}}\,2^{-k};
\]
Cauchy--Schwarz applies because the squares are respectively an exponential
series and a geometric series.  Consequently the same constant controls every
finite subset of HLP levels, and passing to the full summable hierarchy gives
the asserted subset-uniform bound. The ordinary $k=0$ coefficient is controlled by
$\sum_{m\le x}\Lambda(m)^2\le (\log x)\psi(x)\ll_{C_x} xL$; fixed coefficient
translations multiply it by
\[
m^{O(1/L)}\le \exp\!\left(O\!\left(\frac{\log(TL^{C_x})}{L}\right)\right)=O_{C_x}(1).
\]
A support restriction only sets
some Hilbert-valued coefficient coordinates to zero and therefore cannot
increase the square sum. This proves \eqref{eq:direct-carrier-square}.
\end{proof}

\begin{lemma}[Uniform freezing of the slow half-self factor]
\label{lem:block-freezing}
Let a smooth height block be supported on
$|t-\tau_b|\ll H=T/L^D$ on the fixed right edge $\Re s=c>1$, and let
$F_b=f(1/2+i\tau_b)$ be as in \eqref{eq:Fb-real-freeze}. Then
\begin{equation}
\boxed{
|f(c+it)-F_b|\ll L^{-D},
\qquad
|f(c+it)^{-k}-F_b^{-k}|
\ll kL^{-k-D-1}\quad(k\ge1).
}
\label{eq:block-freezing-coeff}
\end{equation}
Independently of any HLP local projection, let $K_{J,\rm var,b}$ and
$K_{J,\rm fr,b}$ denote the normalized critical-$J$ self lag kernels with
coefficients $f(c+it)$ and $F_b$, respectively.  For the fixed packet, shell,
and curvature derivatives used below there is a fixed $C_{\rm fr}$ such that
\begin{equation}
\boxed{
\partial_\rho^r\partial_p^q
\bigl(K_{J,\rm var,b}-K_{J,\rm fr,b}\bigr)
=O\!\left(L^{-D-1+C_{\rm fr}}\right),
\qquad 0\le r,q\le2.
}
\label{eq:self-freezing-kernel}
\end{equation}
The same estimate holds for its diagonal derivative jump.

For the optional stronger comparison, after restricting the HLP hierarchy to
$k\le K_0$ and applying the local principal projector, the normalized local
kernels obtained from \eqref{eq:Hex-variable} and \eqref{eq:Hex} satisfy,
for the derivative orders used below,
\begin{equation}
\boxed{
\partial_\rho^r\partial_p^q
\bigl(K_{\rm var,b}-K_{\rm fr,b}\bigr)
=O\!\left(L^{-D-1+C_{\rm fr}}\right),
\qquad 0\le r,q\le2.
}
\label{eq:block-freezing-kernel}
\end{equation}
The same estimate holds for the corresponding diagonal derivative jump.
\end{lemma}

\begin{proof}
Stirling's formula on a fixed vertical line gives
$f(c+it)=\frac12\log(t/2\pi)+O(T^{-1})$ and
$\partial_t f(c+it)=O(T^{-1})$. Moreover $f'(s)=O(T^{-1})$ uniformly on the
bounded horizontal segment joining $1/2+i\tau_b$ to $c+i\tau_b$. Therefore
\[
|f(c+it)-F_b|
\ll \frac{|t-\tau_b|}{T}+T^{-1}
\ll L^{-D}.
\]
Since both $f(c+it)$ and $F_b$ have modulus
$\asymp L$, the mean-value theorem applied to $z\mapsto z^{-k}$ gives the
second estimate in \eqref{eq:block-freezing-coeff}.

The normalized critical-$J$ self term contains the scalar factor $2f/L$, so
its freezing error is $O(L^{-D-1})$.  The fixed packet, shell, endpoint, and
curvature derivatives cost only $L^{C_{\rm fr}}$ and do not alter the
one-sided packet factorization; this proves \eqref{eq:self-freezing-kernel} and
the corresponding jump estimate.

For the optional locally projected HLP comparison, at level $k$ every
occurrence of the slow factor is
multiplied by $f^{-k}$; after a fixed number of stationary and packet
derivatives the loss is bounded by
$L^{C_{\rm fr}}(1+k)^{C_{\rm fr}}$. Proposition~\ref{prop:factorial-majorant}
therefore bounds the required sum uniformly in $K_0$, since
\[
\sum_{1\le k\le K_0}
\frac{R^k(1+k)^{C_{\rm fr}+1}}{\sqrt{k!}}
\le
\sum_{k\ge1}
\frac{R^k(1+k)^{C_{\rm fr}+1}}{\sqrt{k!}}<\infty.
\]
The finite Cauchy and curvature factors have uniform derivatives on the fixed
normalized shift polydisc. This proves
\eqref{eq:block-freezing-kernel}, including the one-sided diagonal derivatives.
\end{proof}

\begin{lemma}[Uniform size of the finite-$T$ model parameters]
\label{lem:principal-parameter-symbol}
For $\tau\in[T,2T]$ and a slow block of length $H=T/L^D$ with reference
height $\tau_b$, put
\[
\lambda(\tau):=\frac{\log(\tau/2\pi)}L,
\qquad
\mu_b:=\frac{2F_b}{L},
\qquad
\delta(\tau):=\lambda(\tau)-1.
\]
Then, uniformly on every retained block,
\begin{equation}
\boxed{
|\lambda(\tau)-1|+|\mu_b-1|+|\delta(\tau)|\ll L^{-1},
\qquad
(\tau\partial_\tau)^j\lambda(\tau)=O_j(L^{-1})\quad(1\le j\le3).
}
\label{eq:principal-field-symbol-bound}
\end{equation}
Moreover
\begin{equation}
\boxed{
\mu_b-\lambda(\tau)
=O(L^{-D-1})+O((TL)^{-1}).
}
\label{eq:principal-mu-lambda-small}
\end{equation}
These are physical-height symbol estimates only.  No assertion is made that
$\lambda(\tau)$, $\mu_b$, or $\delta(\tau)$ becomes a scalar function of the
final two lag variables after stationary straightening.
\end{lemma}

\begin{proof}
Since $L=\log(T/2\pi)$,
\[
\lambda(\tau)-1=\frac{\log(\tau/T)}L=O(L^{-1})
\]
uniformly for $T\le\tau\le2T$, and direct differentiation gives the displayed
$\tau\partial_\tau$ bounds.  Stirling's formula for the half-self factor gives
\[
\frac{2f(1/2+i\tau_b)}L
=\frac{\log(\tau_b/2\pi)}L+O((TL)^{-1}).
\]
Because $|\tau-\tau_b|\ll H=T/L^D$, the mean-value theorem yields
\[
\frac{\log(\tau_b/2\pi)-\log(\tau/2\pi)}L
=O\!\left(\frac{H}{TL}\right)=O(L^{-D-1}),
\]
which proves \eqref{eq:principal-mu-lambda-small} and hence
\eqref{eq:principal-field-symbol-bound}.  The final sentence is deliberate:
physical height is still tied to the stationary carrier, so the later proof
places these finite-$T$ deformations in the regular ledger rather than using a
variable-coefficient positivity transfer.
\end{proof}

\begin{lemma}[High exact HLP levels remain direct sideband terms]
\label{lem:HLP-high-edge}
Let $K_0=L^{1/8}$. Keep the stationary levels $k>K_0$ in their original
pointwise-$f(t)$, absolutely convergent safe-line Dirichlet representation; no
block freezing or local Perron projector is applied to them. On a retained unit shell put
$d_j\asymp T/J_j$ and
\[
R_j^{\rm sb}:=\epsilon_Ld_j,
\]
with $\epsilon_L=L^{-\sigma}$ fixed in Section~\ref{sec:retained}. Expand the
smooth stationary amplitude in shell Fourier modes $r$. The part
$|r|\le R_j^{\rm sb}$ and its adjoint define $E_T^{\rm HLP,hi}$ with
\begin{equation}
\boxed{
\rank E_T^{\rm HLP,hi}\ll T=o(N_T).
}
\label{eq:HLP-high-edge}
\end{equation}
Write $R_T^{\rm HLP,hi}$ for the complementary part
$|r|>R_j^{\rm sb}$; it is disjoint from $E_T^{\rm HLP,hi}$ and is estimated
later as regular class~(vii).
\end{lemma}

\begin{proof}
The safe-line geometric expansion converges absolutely. Each level changes only
the coefficient of the total index $M$, and
Lemma~\ref{lem:hlp-radial-variable} gives carrier $e(-My)$ with
$t_*=2\pi My$. Lemma~\ref{lem:stationary-symbol} is uniform over the summed
exact HLP hierarchy, so its shell amplitude has the same Fourier-sideband decay
used in class~(vii). The low sidebands replace $M$ by the effective carrier
$M-r$ and hence lie in the fixed edge spaces of
Proposition~\ref{prop:edge-rank}. By Proposition~\ref{prop:trivial-outer-twist}
there is no additional arithmetic multiplicity, which gives
\eqref{eq:HLP-high-edge}. The complementary sidebands are exactly the
high-sideband class~(vii).
\end{proof}

\begin{proposition}[Exact--model difference remains a direct one-$\chi$ family]
\label{prop:direct-exact-model-residual}
On a retained height block with reference value $F_b$, define on the fixed
absolute-convergence right edge
\begin{equation}
\boxed{
\Delta H_{\rm dir}(s)
:=P(s)+\frac{Q(s)}{f(s)-P(s)}-P(s-2/L).
}
\label{eq:direct-exact-model-residual}
\end{equation}
Then the exact pointwise principal source satisfies the literal identity
\begin{equation}
\boxed{
\widehat H_{\rm ex}(s)
=H_{\rm mod}(s;F_b)
+\Delta H_{\rm dir}(s)
+\bigl(f(s)-F_b\bigr).
}
\label{eq:direct-model-source-split}
\end{equation}
After multiplication by the common shifted-curvature source and the single
functional-equation factor, every stationary term generated by
$\Delta H_{\rm dir}$ is a direct physical one-$\chi$ carrier.  Its complete
HLP hierarchy is norm-summable in the direct-carrier square bound
\eqref{eq:direct-carrier-square}.  Consequently its low Fourier sidebands lie
in the common edge space of Proposition~\ref{prop:edge-rank}, with total rank
$O(T)=o(N_T)$.  Its complementary high sidebands retain the uniform coefficient
square-sum and smooth-symbol bounds needed for the direct high-sideband estimate
in Section~\ref{sec:ledger}.  The last term in
\eqref{eq:direct-model-source-split} is a slow self-freezing error and is
relative-regular.
\end{proposition}

\begin{proof}
Subtract \eqref{eq:Hmod} from \eqref{eq:Hex-variable}; the displayed identity
is immediate.  On the safe line the geometric series
\[
\frac{Q}{f-P}=\sum_{k\ge1}f^{-k}P^{k-1}Q
\]
is absolutely and locally uniformly convergent.  By
Lemma~\ref{lem:hlp-radial-variable}, every level changes only the coefficient
of the total positive Dirichlet index $M$ and retains the saddle and phase
\[
t_*=2\pi My,\qquad e(-My).
\]
The two remaining pieces $P(s)$ and $P(s-2/L)$ are ordinary direct sources; the
translation multiplies their coefficient at $M$ by $M^{2/L}=O(1)$ and creates
no new carrier.  Lemma~\ref{lem:direct-carrier-square} applies to the full
norm-summable hierarchy, and Lemma~\ref{lem:stationary-symbol} supplies uniform
Fourier-sideband decay.  Proposition~\ref{prop:edge-rank} is independent of
coefficients and signs and also applies to norm-convergent sums, giving the
low-sideband rank assertion.  For the complementary sidebands the same two
bounds---the direct coefficient square sum and the arbitrary fixed-order
Fourier decay of the stationary symbol---are retained for the later
Hilbert--Schmidt summation.

Finally, the unconditional self estimate
\eqref{eq:self-freezing-kernel} gives the normalized critical-$J$ discrepancy
from $f-F_b$, including its fixed endpoint/packet derivatives, as
$O(L^{-D-1+C_{\rm fr}})=o(1)$.  It therefore satisfies the primitive regular
criterion.  No HLP level
is frozen or locally projected in this decomposition.
\end{proof}

\subsection{Optional exact-HLP local comparison and the smooth packet Mellin class}
\label{sec:local-projector}

The remainder of this subsection records a stronger levelwise exact-HLP local
comparison.  It is useful as an independent consistency check, but the final
source decomposition of Proposition~\ref{prop:uniform-twisted-replacement}
does not invoke it: in the proof of Theorem~\ref{thm:main} all exact HLP levels
remain on the original absolute-convergence right edge.

The principal local projection in this optional comparison is defined levelwise. Expand the exact
resolvent geometrically on the absolute-convergence right edge and let
$\mathcal F_{k,L}(1+u;p)$ denote one completed retained level after the finite
three-$Z$/Cauchy completion has been formed. Its common principal local singularity is the complete Laurent part at
$u=0$, including the actual multiplicity created by coalescing faces. Define
\begin{equation}
\Pi_0^{(k)}\mathcal F_{k,L}
:=\operatorname{PP}_{u=0}\mathcal F_{k,L}.
\label{eq:levelwise-PP-projector}
\end{equation}
For bookkeeping in a general completed family, if that level has additional
translated one-$\chi$ poles separated from the common local cluster, let
$\Sigma_{{\rm tr},k}$ be their finite set. Its cardinality is bounded
independently of $k$: such poles can arise only from the fixed three-$Z$/Cauchy
completion, while the HLP level changes the pole multiplicity at the common
point $u=0$.  Lemma~\ref{lem:cluster-levelwise} will show that in the
invariant-source specialization used in this paper this auxiliary set is
actually empty. Put
\begin{equation}
\Pi_{\Sigma_{{\rm tr},k}}F
:=\sum_{u_0\in\Sigma_{{\rm tr},k}}
\operatorname{PP}_{u=u_0}F.
\label{eq:local-PP-projector}
\end{equation}
Then the exact level has the mutually exclusive decomposition
\begin{equation}
\boxed{
\mathcal F_{k,L}
=\Pi_0^{(k)}\mathcal F_{k,L}
+\Pi_{\Sigma_{{\rm tr},k}}\mathcal F_{k,L}
+\mathcal F_{k,L}^{\rm hol},
}
\label{eq:canonical-local-decomp}
\end{equation}
where $\mathcal F_{k,L}^{\rm hol}$ is holomorphic throughout the local
neighbourhood. Identity \eqref{eq:canonical-local-decomp} is algebraic for every
$k$, but the exact source decomposition applies its local projectors only for
$k\le K_0$. Thus the exact locally projected functional is the finite sum
\begin{equation}
\mathcal L_{\rm pr,\le K_0}^{\rm ex}[W]
:=\sum_{0\le k\le K_0}\omega_{k,L}
\Res_{u=0}\!\bigl(\mathcal F_{k,L}(1+u;p)W(u)\bigr).
\label{eq:truncated-exact-principal-functional}
\end{equation}
The scalar level weight contains the frozen factor $F_b^{-k}$ together with
only fixed normalized completion factors. Since $F_b\asymp L$, there is a fixed
constant $C_0$ such that
\begin{equation}
\boxed{\omega_{k,L}\ll C_0^kL^{-k}.}
\label{eq:level-weight-bound}
\end{equation}
The levels $k>K_0$ are not projected and remain in the direct sideband
decomposition of Lemma~\ref{lem:HLP-high-edge}.

For comparison with the positive model, replace the leading Laurent factor at
each level and resum the resulting \emph{model} hierarchy over all $k$. This is
not a second extraction of the exact high-level source. Lemma~\ref{lem:leading-jet-resummation}
then produces the model cluster
\begin{equation}
\boxed{\Sigma_0^{\rm mod}=\{0,2/L\},}
\label{eq:model-principal-cluster}
\end{equation}
with residues $-2,+1$. For the resummed model define
\begin{equation}
\Pi_{\Sigma_0^{\rm mod}}F
:=\sum_{u_0\in\{0,2/L\}}\operatorname{PP}_{u=u_0}F.
\label{eq:model-PP-projector}
\end{equation}
Thus the translated model pole belongs to the \emph{model} principal
projector, while the algebraic point $u_*(F_b)$ of
Lemma~\ref{lem:principal-pole-cluster} serves only to identify the same
translated scale.  In a general completed family the bookkeeping
\eqref{eq:canonical-local-decomp} separates any genuinely additional exact
poles from the common cluster and the holomorphic remainder.  In the
invariant-source specialization the actual projector is applied only after the
curvature coefficients $q\le2$ have been formed at $p=0$; for those
coefficients Lemma~\ref{lem:cluster-levelwise} shows that the additional
exact-pole slot is empty.

\begin{lemma}[Smooth bounded-support Mellin weights]
\label{lem:smooth-mellin-class}
Let $0\le t\le C_*$ with $C_*$ fixed. Assume that for some fixed derivative
order $R_*$, large enough for the applications below,
\[
\|\partial_t^m\partial_\rho^r\partial_p^j a_{\rho,p,T}\|_{L^1(0,C_*)}
\le L^{A_*+o(1)}
\qquad (m+r+j\le R_*).
\]
Define
\begin{equation}
W_{L,\rho,p}(u)
:=\int_0^{C_*}a_{\rho,p,T}(t)e^{Lut}\,dt
=\sum_{n\ge0}w_n(\rho,p)u^n.
\label{eq:smooth-mellin-weight}
\end{equation}
Then
\begin{equation}
\boxed{
|\partial_\rho^r\partial_p^j w_n(\rho,p)|
\le L^{A_*+o(1)}\frac{(C_*L)^n}{n!}.
}
\label{eq:smooth-mellin-factorial}
\end{equation}
The same conclusion holds for a moving endpoint
$\int_0^\rho a_{\rho,p,T}(t)e^{Lut}\,dt$, and the class is stable under every
fixed finite product of such weights.
\end{lemma}

\begin{proof}
Expanding the exponential and using Fubini gives
\[
w_n(\rho,p)=\frac{L^n}{n!}
\int_0^{C_*}a_{\rho,p,T}(t)t^n\,dt,
\]
which proves \eqref{eq:smooth-mellin-factorial}. The one-dimensional
$W^{1,1}$ inequality on the fixed interval controls every endpoint value by the
corresponding $L^1$ norm together with one $t$-derivative. Repeated
$\rho$-differentiation of a moving endpoint therefore produces only finitely
many controlled endpoint jets, beginning with
\[
a_{\rho,p,T}(\rho)\frac{(L\rho)^n}{n!},
\]
and integrals of the same form. Finally, products correspond to convolution
of the compactly supported $t$-amplitudes,
so the support remains bounded and the same factorial estimate holds with a
larger fixed $C_*$. \end{proof}

\begin{lemma}[Exact HLP levels preserve the packet radial variable]
\label{lem:hlp-radial-variable}
On the absolute-convergence right edge, multiplication of finitely many
Dirichlet series changes only the coefficient of the total Dirichlet index
$M$. After insertion of the physical packet lag and the outer-denominator
dilation, the corresponding one-$\chi$ term has saddle and phase
\[
t_*=2\pi My,\qquad e(-My).
\]
Thus exact HLP levels do not create an additional packet-support coordinate;
their packet/common-prefix weights belong to the class of
Lemma~\ref{lem:smooth-mellin-class}.
\end{lemma}

\begin{proof}
If $A_1(s)\cdots A_r(s)=\sum_Mc_MM^{-s}$, then after the exact logarithmic
translation the $s$-dependence is $(Me^{-u})^{-s}$ up to a scalar amplitude.
The one-$\chi$ stationary calculation therefore depends on the single total
index $M$, giving the displayed saddle and phase. A translated factor such as
$P(s-2/L)$ merely multiplies its coefficient at $M$ by $M^{2/L}$ and does not
change the stationary variable. \end{proof}

\begin{lemma}[Zero-free control of the unshifted HLP factors]
\label{lem:HLP-zero-free-derivatives}
Let
\[
P(s):=-\frac{\zeta'}{\zeta}(s),
\qquad
Q(s):=-P'(s).
\]
Fix a derivative order $m_*$ and constants $B,C_*>0$. There are constants
$c_{\rm zf},C_{m_*,B,C_*}>0$ such that, with
\[
\eta_L:=\frac{c_{\rm zf}}{\log_3T},
\qquad
V_B:=B\sqrt{\log L},
\]
the rectangle
\[
-\eta_L\le\Re u\le \frac{C_*}{L},
\qquad |\Im u|\le V_B,
\]
contains no zero of $\zeta(1+u)$ and no singularity of the unshifted HLP
factors except the pole at $u=0$. Moreover, for every $0\le m\le m_*$,
\begin{equation}
\boxed{
\partial_u^m P(1+u)
=\frac{(-1)^m m!}{u^{m+1}}+R_m(u),
\qquad
|R_m(u)|\ll_{m,B} (\log_3T)^{m+1},
}
\label{eq:HLP-zero-free-derivatives}
\end{equation}
uniformly on the rectangle. Consequently every fixed derivative of $Q(1+u)$
is the corresponding principal Laurent term plus a fixed power of
$\log_3T$. In particular, on the left and horizontal sides of the rectangle,
all fixed derivatives of $P$ and $Q$ are $O((\log_3T)^C)$ for a fixed $C$.
\end{lemma}

\begin{proof}
Use the classical zero-free region
\[
\zeta(\sigma+it)\ne0
\qquad
\left(\sigma\ge1-\frac{c}{\log(|t|+3)}\right)
\]
with a sufficiently small absolute $c>0$; see, for example,
\cite{Titchmarsh}. Since $|t|\le V_B$ and
$\log(V_B+3)\ll_B\log_3T$, reducing $c_{\rm zf}$ if necessary places the whole
rectangle inside this zero-free region, apart from the pole of $\zeta$ at
$s=1$. The standard logarithmic-derivative estimate in the same region gives
\[
P(1+u)-u^{-1}=O(\log_3T).
\]
The difference is holomorphic on a slightly enlarged rectangle. Cauchy's
estimate on circles of radius comparable to $1/\log_3T$ therefore gives
\[
\partial_u^m\bigl(P(1+u)-u^{-1}\bigr)
\ll_m(\log_3T)^{m+1}
\]
for every fixed $m\le m_*$. This proves
\eqref{eq:HLP-zero-free-derivatives}. Since $Q=-P'$, the assertions for $Q$
follow immediately. On the left side $|u|\gg1/\log_3T$, and on the horizontal
sides $|u|\gg1$, so the explicit Laurent terms satisfy the same fixed-power
bound there.
\end{proof}

\begin{lemma}[Uniform boundary bound for the low HLP hierarchy]
\label{lem:HLP-boundary-hierarchy}
Fix $A,C_H>0$ and choose $C_*>4C_0$ sufficiently large. Apply
Lemma~\ref{lem:HLP-zero-free-derivatives} with $B=C_H\sqrt A$, let
$\eta_L=c_{\rm zf}/\log_3T$ be the resulting zero-free width, and put
\[
V_A:=C_H\sqrt{A\log L},
\qquad
\mathscr R_{L,A}:=
\{u:-\eta_L\le\Re u\le C_*/L,\ |\Im u|\le V_A\}.
\]
For $0\le q\le2$ put
\[
\mathcal F_{k,L}^{[q]}(1+u)
:=\left.\partial_p^q\mathcal F_{k,L}(1+u;p)\right|_{p=0}.
\]
On the boundary of $\mathscr R_{L,A}$, after the common fixed completion
factors are included but before the packet Mellin weight is applied, there is a
fixed exponent $C_{\rm HLP}$ such that
\begin{equation}
\boxed{
\sum_{0\le k\le K_0}
|\omega_{k,L}|\,
|\mathcal F_{k,L}^{[q]}(1+u)|
\ll L^{C_{\rm HLP}},
\qquad 0\le q\le2.
}
\label{eq:HLP-boundary-hierarchy}
\end{equation}
The same estimate holds after any of the fixed normalized three-$Z$ shift
derivatives used in the principal calculation.
\end{lemma}

\begin{proof}
The $k$-dependent part before completion is $P^{k-1}Q$ for $k\ge1$; the frozen
factor $F_b^{-k}$ and the remaining fixed scalar completion factors are
included in $\omega_{k,L}$, which satisfies
\eqref{eq:level-weight-bound}.  A fixed curvature or normalized shift
derivative produces only a fixed-degree polynomial in $k$ and replaces a fixed
number of $P,Q$ factors by fixed derivatives of them.  On the left and
horizontal sides, Lemma~\ref{lem:HLP-zero-free-derivatives} bounds every such
factor by a fixed power of $\log_3T$. Combining this with
\eqref{eq:level-weight-bound} gives a geometric majorant whose successive ratio
is $O((\log_3T)^C/L)=o(1)$. On the right side,
\eqref{eq:HLP-zero-free-derivatives} gives $|P(1+u)|\le L/C_*+L^{o(1)}$.
Hence the corresponding geometric majorant has successive ratio at most
$C_0/C_*+o(1)$; by the choice of $C_*$ this is bounded by a fixed number $<1$.
The finitely many extra derivatives contribute only polynomial factors in $k$,
which remain summable against this geometric majorant.  The ordinary $k=0$ face and all fixed completion factors cost
only a fixed power of $L$ by the same Cauchy estimates used in
Lemma~\ref{lem:packet-HLZ-diagonal}.  Summing the hierarchy proves
\eqref{eq:HLP-boundary-hierarchy}.
\end{proof}

\begin{lemma}[Levelwise Perron deformation after the Dirichlet sum]
\label{lem:cluster-levelwise}
Assume Hypothesis~\ref{hyp:packet-bilinear-HLZ}(i), so that the completed
coefficients below are the Cauchy transforms of the actual right-edge source.
Fix $C_*>4C_0$ sufficiently large and choose $C_H>0$ sufficiently large for
the Gaussian decay of the explicit separated-torus factors
$H_{\mathbf z}$ from \eqref{eq:HLZ-explicit-auxiliary}. Put
\[
\Gamma_L:=\{u:|u|=C_*/L\}.
\]
For every fixed $A>0$, let $\eta_L,V_A,\mathscr R_{L,A}$ and the curvature
coefficients $\mathcal F_{k,L}^{[q]}$ be those of
Lemma~\ref{lem:HLP-boundary-hierarchy} with these same constants $C_*,C_H$.
For the invariant-source specialization and every $k\le K_0$, form the exact
Dirichlet series for the $k$th HLP level first on a fixed safe line
$\Re u=\delta>0$, where it is absolutely convergent, and only then identify its
sum with the corresponding completed meromorphic function
$\mathcal F_{k,L}(1+u;p)$.
After the fixed three-$Z$/Cauchy completion, each coefficient
$\mathcal F_{k,L}^{[q]}$ has in $\mathscr R_{L,A}$ only the common pole at
$u=0$, with the higher order allowed
by the coalescing differentiated shifts. For each $q\le2$, let $\Sigma_{{\rm tr},k}^{[q]}$ denote the additional-pole
set of the curvature coefficient $\mathcal F_{k,L}^{[q]}$.  Then for the exact
curvature coefficients actually used in the invariant-source low-level
projection,
\begin{equation}
\boxed{\Sigma_{{\rm tr},k}^{[q]}=\varnothing,
\qquad 0\le q\le2.}
\label{eq:no-exact-translated-poles}
\end{equation}
The point $2/L$ is a pole only of the \emph{resummed leading model} of
Lemma~\ref{lem:leading-jet-resummation}, not of an individual exact HLP
curvature coefficient.

Let $W_p$ be any packet Mellin weight of Lemma~\ref{lem:smooth-mellin-class},
with the allowed smooth $p$-dependence, and define the levelwise Perron integral
on the original absolute-convergence line by
\[
I_{k,L}^{\rm orig}[W_p]
:=\frac1{2\pi i}\int_{\delta-i\infty}^{\delta+i\infty}
W_p(u)\mathcal F_{k,L}(1+u;p)\,du,
\]
with the separated Cauchy-torus integration and the factor
$H_{\mathbf z}$ understood as part of the completed level.
After the common $L^2$ Perron normalization, for
$0\le q\le2$ and uniformly for the fixed $\rho$-derivatives used in the paper,
\begin{equation}
\boxed{
\left.\partial_p^q\left[
\sum_{0\le k\le K_0}\omega_{k,L}I_{k,L}^{\rm orig}[W_p]
-
\sum_{0\le k\le K_0}\omega_{k,L}
\Res_{u=0}\!\bigl(W_p(u)\mathcal F_{k,L}(1+u;p)\bigr)
\right]\right|_{p=0}
=O_A(L^{-A}).
}
\label{eq:levelwise-Perron-deformation}
\end{equation}
The Dirichlet series and the contour integral are therefore interchanged only
on the original half-plane of absolute convergence. No Dirichlet series is
used on the part of the deformation with $\Re(1+u)<1$.
\end{lemma}

\begin{proof}
Choose first a fixed safe line $\Re u=\delta>0$. On that line every Dirichlet
series entering the completed $k$th level is absolutely and locally uniformly
convergent for bounded $p$. Hence, for each $0\le q\le2$, first take
$\partial_p^q$ under the Dirichlet sums and the original Perron integral and
then set $p=0$. Only after this curvature coefficient has been formed is the
Perron contour moved. The internal HLP sum at $p=0$ gives $P^{k-1}Q$, with the
fixed number of curvature derivatives producing only finite products of its
unshifted derivatives; the scalar factor of size $F_b^{-k}$ is carried by the
level weight $\omega_{k,L}$. Thus the function to be deformed is the
already-summed meromorphic coefficient $\mathcal F_{k,L}^{[q]}(1+u)$, not a
$p\ne0$ Dirichlet series. As in the proof of the HLZ diagonal lemma, this
summed coefficient is holomorphic between the fixed safe line and
$\Re u=C_*/L$, so the first move crosses no pole.

The remaining three-$Z$ shift variables are handled on the separated Cauchy
torus \eqref{eq:HLZ-separated-circles}.  For each
$\mathbf z\in\mathcal T_L$, the explicit factor
$H_{\mathbf z}$ cancels the two moving divisors
$2u=z_2-z_1$ and $2u=z_2-z_3$ pointwise by
\eqref{eq:HLZ-auxiliary-properties}.  The physical shift coefficients are then
taken by the outer finite Cauchy integrals; no jointly analytic auxiliary
factor through $z_1=z_2=z_3$ is invoked. Apply
Lemma~\ref{lem:HLP-zero-free-derivatives} with $B=C_H\sqrt A$ and derivative
order larger than every HLP derivative produced by the fixed curvature
extraction.  After the moving divisors are cancelled, the remaining unshifted
$P,Q$ factors have no singularity in the deformation rectangle except their
Laurent pole at $u=0$.  The finite Cauchy integration in $\mathbf z$ cannot
create a new $u$-pole because its contours are fixed and separated from their
own finite poles.  Hence $\mathcal F_{k,L}^{[q]}$ has only the coalesced common
Laurent pole at $u=0$, proving \eqref{eq:no-exact-translated-poles}. By
contrast, the point $2/L$ arises only after the infinite leading Laurent
hierarchy is geometrically resummed, as shown in
\eqref{eq:leading-geometric-resummation}.

It remains to justify the uniform deformation.  By
Lemma~\ref{lem:HLP-boundary-hierarchy}, after summing all $k\le K_0$ the
completed HLP curvature coefficients have size at most $L^{C_{\rm HLP}}$ on
the whole boundary of $\mathscr R_{L,A}$.  Lemma~\ref{lem:smooth-mellin-class}
gives the same fixed-power bound for $W_p$ and the required derivatives.
Thus, apart from the explicit Perron and Gaussian factors, every boundary
integrand produced by the fixed Leibniz expansion is bounded by one fixed power
of $L$.

Now repeat the one-variable deformation used in
Lemma~\ref{lem:packet-HLZ-diagonal}, separately for each total curvature
coefficient obtained by Leibniz' rule from
$W_p\mathcal F_{k,L}(1+u;p)$ at $p=0$: truncate at $|\Im u|=V_A$, move the
summed coefficient to $\Re u=-\eta_L$, and apply Cauchy's theorem. The
completed level retains the same diagonal Perron factor $T^{2u}$ (up to the
harmless fixed finite-fibre normalization) as the HLZ diagonal. The Gaussian decay of the separated-torus factor
$H_{\mathbf z}$ makes the truncation and horizontal connectors
$O_A(L^{-A})$, uniformly in $\mathbf z\in\mathcal T_L$, by the choice of
$C_H$, while on the left vertical $T^{2u}$ supplies
$T^{-2\eta_L}$ and the summed HLP hierarchy costs only the fixed
polylogarithmic factor established above. Thus the left vertical has the same
superlogarithmic saving class as \eqref{eq:packet-HLZ-error}. The only crossed
pole is $u=0$. Summing the residue identities over $k\le K_0$ and then
reassembling the Leibniz terms gives \eqref{eq:levelwise-Perron-deformation}.
All operations to the left of $\Re u=0$ are performed on the summed
meromorphic curvature coefficients, never on their Dirichlet series.
\end{proof}

\begin{lemma}[Leading HLP jet resums to the translated model cluster]
\label{lem:leading-jet-resummation}
At the principal Laurent level
\[
P(1+u)\rightsquigarrow u^{-1},
\qquad Q(1+u)\rightsquigarrow u^{-2},
\]
the complete geometric hierarchy in $Q/(F_b-P)$ resums exactly as
\begin{equation}
\boxed{
\sum_{k\ge1}F_b^{-k}u^{-k-1}
=-\frac1u+\frac1{u-F_b^{-1}}.
}
\label{eq:leading-geometric-resummation}
\end{equation}
Consequently the \emph{resummed leading Laurent family} of the completed
source has generating principal part
\begin{equation}
\boxed{-\frac2u+\frac1{u-F_b^{-1}}.}
\label{eq:leading-completed-cluster}
\end{equation}
Since $F_b=L/2+O(1)$,
\[
F_b^{-1}=\frac2L+O(L^{-2}),
\]
so \eqref{eq:leading-completed-cluster} is precisely the finite-$T$ precursor
of the model principal parts
$-2/u+1/(u-2/L)$ in \eqref{eq:Hmod}. The $O(L^{-2})$ displacement of the
translated pole is one-log lower after the common $L^2$ Perron normalization
and is included in the exact--model local comparison below.
\end{lemma}

\begin{proof}
The geometric expansion on $\Gamma_L$ is
\[
\frac{Q}{F_b-P}=\sum_{k\ge1}F_b^{-k}QP^{k-1}.
\]
Replacing only the common principal Laurent factors gives the series on the
left of \eqref{eq:leading-geometric-resummation}. Summing it as a geometric
series gives
\[
\frac1{F_bu^2}\frac1{1-(F_bu)^{-1}}
=\frac1{u(F_bu-1)}
=-\frac1u+\frac1{u-F_b^{-1}}.
\]
Adding the principal part $-u^{-1}$ of the separate $-P$ source proves
\eqref{eq:leading-completed-cluster}. Multiplication by the common shifted-zeta,
Cauchy, curvature, and packet factors does not alter this source-level pole
identity.
\end{proof}

Now write the independent Perron variable as $z=1+u$. Put
\[
A(u):=uP(1+u),\qquad
B(u):=u^2Q(1+u),\qquad
C(u):=u^2\zeta(1+u)^2.
\]
On a fixed disk $|u|\le r_0$, the Laurent expansions at the zeta pole show that
$A,B,C$ are analytic, $A(0)=B(0)=C(0)=1$, and all fixed derivatives are
bounded. For $k\ge1$ and $0\le q\le2$, differentiate the completed level $q$
times with respect to the normalized curvature shift $p$ and then set $p=0$.
A derivative falling on a factor shifted by $a=p/L$ contributes one factor
$L^{-1}$ and can raise the local pole order by one; derivatives falling on the
normalized archimedean or packet factors do not raise the pole order and need
not carry this factor. In particular the twice pole-raising contribution is
fixed by the exact curvature identity \eqref{eq:curvature-source}:
\begin{equation}
\boxed{
L^{-2}Q(1+u)^2P(1+u)^{k-1}\zeta(1+u)^2
=L^{-2}u^{-k-5}A(u)^{k-1}B(u)^2C(u).
}
\label{eq:uniform-level-factorization}
\end{equation}
The fixed shift residues commute with these derivatives by
\eqref{eq:residue-commute}. After the moving shift divisors have been canceled,
Leibniz' rule and Cauchy's estimate therefore give, for $k\le K_0$,
\begin{equation}
\boxed{
\left.\partial_p^q\mathcal F_{k,L}^{\rm ex}(1+u;p)\right|_{p=0}
=\sum_{\ell=0}^qL^{-\ell}u^{-k-3-\ell}
\bigl(H_{k,q,\ell,0}+u\Theta_{k,q,\ell,L}(u)\bigr),
\qquad 0\le q\le2.
}
\label{eq:level-Laurent}
\end{equation}
Here $\ell$ records the number of derivatives that actually raise the local
pole order. For every fixed $a$ used below,
\begin{equation}
\boxed{
|H_{k,q,\ell,0}|+|\partial_u^a\Theta_{k,q,\ell,L}(u)|
\ll L^{o(1)}C_H^k(1+k)^C,
\qquad 0\le\ell\le q\le2.
}
\label{eq:uniform-Hk}
\end{equation}
The separate ordinary face $k=0$ has fixed pole order and is covered by the
same finite estimate after enlarging the constant. The levels $k>K_0$ are not
locally projected, by Lemma~\ref{lem:HLP-high-edge}. Hence the constants in the
local Laurent error are uniform through $k\le L^{1/8}$; no fixed-$k$ asymptotic
is being summed. Package the three required curvature coefficients into the
order-two intermediate leading Taylor jet
\begin{equation}
\boxed{
\mathcal F_{k,L}^{\rm lead,(2)}(1+u;p)
:=\sum_{q=0}^2\frac{p^q}{q!}
\sum_{\ell=0}^qL^{-\ell}u^{-k-3-\ell}H_{k,q,\ell,0}.
}
\label{eq:intermediate-leading-jet}
\end{equation}
Its derivatives through order two at $p=0$ are exactly the leading terms in
\eqref{eq:level-Laurent}. This jet is not resummed from arbitrary
$k$-dependent coefficients: it is obtained by taking the same finite
$p$-derivatives of the absolutely convergent leading geometric family in
Lemma~\ref{lem:leading-jet-resummation}. Hence differentiation and geometric
resummation commute, and the translated-pole parameter remains the one given
there. No higher $p$-jet is used in the proof.  We use the level-weight bound
\eqref{eq:level-weight-bound} below. For $k\le L^{1/8}$, write
\[
W_{L,\rho,p}(u)=\sum_{n\ge0}w_n(\rho,p)u^n.
\]
The finite-interval Laplace representation gives, for $r,j\le2$,
\begin{equation}
|\partial_\rho^r\partial_p^j w_n(\rho,p)|
\le C L^nB^n\frac{(1+n)^C}{n!}.
\label{eq:packet-weight-bound}
\end{equation}
Define the packet-local residue before curvature extraction by
\[
\mathcal L_{k,L}^{\rm pkt}(\rho,p)
:=\omega_{k,L}\Res_{u=0}
\bigl[\mathcal F_{k,L}(1+u;p)W_{L,\rho,p}(u)\bigr].
\]
All $p$-derivatives in the estimates below are evaluated at $p=0$. For a term
in $\partial_p^q\mathcal L_{k,L}^{\rm pkt}$ with $q_1$ derivatives falling on
$\mathcal F_{k,L}$ and $q-q_1$ on $W$, decompose the first factor according to
\eqref{eq:level-Laurent}. Its error part is a finite sum over
$0\le\ell\le q_1$ of
\[
\omega_{k,L}L^{-\ell}\Res_{u=0}
\left[u^{-k-2-\ell}\Theta_{k,q_1,\ell,L}(u)
\,\partial_p^{q-q_1}W_{L,\rho,p}(u)\big|_{p=0}\right].
\]
Expanding $\Theta_{k,q_1,\ell,L}=\sum_j\theta_{k,q_1,\ell,j}u^j$ and using
\eqref{eq:uniform-Hk} and \eqref{eq:packet-weight-bound} gives, for each
$\ell$,
\[
\ll L^{-k-\ell+o(1)}C^k(1+k)^C
\sum_{j=0}^{k+1+\ell}r_0^{-j}
\frac{(BL)^{k+1+\ell-j}(1+k)^C}
{(k+1+\ell-j)!}.
\]
Because $\ell\le q_1\le2$ is fixed and $k/L=o(1)$, the same factorial
comparison as above makes the sum uniformly bounded after factoring its $j=0$
term. Hence, uniformly for $r,q\le2$,
\begin{equation}
\boxed{
|\partial_\rho^r\partial_p^q\Delta\mathcal L_{k,L}^{\rm pkt}(\rho,0)|
\ll L^{1+o(1)}
\frac{C_1^k(1+k)^{C_1}}{(k+1)!}.
}
\label{eq:level-local-error}
\end{equation}
Only the levels $k\le K_0$ enter this local comparison. By
Lemma~\ref{lem:HLP-high-edge}, the higher exact levels retain their direct
one-$\chi$ representation: their low sidebands belong to
$E_T^{\rm HLP,hi}$ and their high sidebands belong to regular class~(vii).

The normalization is now explicit. For the component indexed by $\ell$, the
leading jet has pole order $k+3+\ell$, so residue extraction uses
$w_{k+2+\ell}$. Its natural size is
\[
L^{-k-\ell}\frac{L^{k+2+\ell}}{(k+2+\ell)!}
=L^2\frac1{(k+2+\ell)!}.
\]
Thus every increase of the local pole order is paired with exactly the same
power of $L^{-1}$ from the normalized shift. In particular, the twice
pole-raising curvature contribution in \eqref{eq:uniform-level-factorization}
uses $w_{k+4}$ together with $L^{-2}$ and still has the common $L^2$ Perron
scale. Let $K_{\rm lead,\le K_0}^{\rm pkt}$ be the sum of these leading jets for
$k\le K_0$. After the common positive Perron amplitude $L^2$ is divided out,
\eqref{eq:level-local-error} gives
\begin{equation}
\partial_\rho^r\partial_p^q
\bigl(K_{\rm ex,\le K_0}^{\rm pkt}-K_{\rm lead,\le K_0}^{\rm pkt}\bigr)
=O(L^{-1+o(1)}),
\qquad r,q\le2,
\label{eq:local-leading-error}
\end{equation}
Here and in all derivative estimates through
\eqref{eq:local-model-error}, the $p$-derivatives are evaluated at $p=0$. Let
$K_{\rm lead,\infty}^{\rm pkt}$ denote the full leading Laurent family before
freezing the translated pole. The same calculation, together with
\eqref{eq:packet-weight-bound} and \eqref{eq:uniform-Hk}, gives
\[
\left|\partial_\rho^r\partial_p^qK_{{\rm lead},k}^{\rm pkt}\right|
\ll \frac{C^k(1+k)^C}{(k+2)!},
\qquad r,q\le2,
\]
after the common $L^2$ normalization.
Therefore, for every fixed $A>0$,
\begin{equation}
\boxed{
\partial_\rho^r\partial_p^q
\bigl(K_{\rm lead,\le K_0}^{\rm pkt}-K_{\rm lead,\infty}^{\rm pkt}\bigr)
=O_A(L^{-A}),
\qquad r,q\le2.
}
\label{eq:model-truncation-tail}
\end{equation}
This is a model-side radial error; no fast one-$\chi$ carrier remains in it.
It remains to compare the full leading family with the fixed translated model. Put
\[
c_b:=\frac{L}{F_b}=\frac{2}{\mu_b}=2+O(L^{-1}).
\]
Lemma~\ref{lem:leading-jet-resummation} shows that the translated pole of the
intermediate leading family is at $u=c_b/L$. After the common $N$-prefix has
been fixed, the resummed leading residue functional is a finite combination of
bounded-support Mellin weights and fixed Cauchy factors whose dependence on
$c_b$ is analytic for $c_b$ in a fixed neighbourhood of $2$; the pole remains a
fixed positive normalized distance from the outer local contour. Lemma
\ref{lem:smooth-mellin-class} and Cauchy's estimate therefore give a uniform
bound for the $c_b$-derivative after up to two $(\rho,p)$ derivatives. The
mean-value theorem gives
\begin{equation}
\partial_\rho^r\partial_p^q
\bigl(K_{\rm lead,\infty}^{\rm pkt}-K_{\rm mod,loc}^{\rm pkt}\bigr)
=O(L^{-1}),
\qquad r,q\le2,
\label{eq:translated-pole-displacement-error}
\end{equation}
where $K_{\rm mod,loc}^{\rm pkt}$ uses the fixed model parameter $c=2$, i.e.
the pole $u=2/L$. The three differences telescope exactly:
\begin{align*}
K_{\rm ex,\le K_0}^{\rm pkt}-K_{\rm mod,loc}^{\rm pkt}
={}&(K_{\rm ex,\le K_0}^{\rm pkt}-K_{\rm lead,\le K_0}^{\rm pkt})\\
&+(K_{\rm lead,\le K_0}^{\rm pkt}-K_{\rm lead,\infty}^{\rm pkt})\\
&+(K_{\rm lead,\infty}^{\rm pkt}-K_{\rm mod,loc}^{\rm pkt}).
\end{align*}
Hence \eqref{eq:local-leading-error}, \eqref{eq:model-truncation-tail}, and
\eqref{eq:translated-pole-displacement-error} give
\begin{equation}
\boxed{
\partial_\rho^r\partial_p^q
\bigl(K_{\rm ex,\le K_0}^{\rm pkt}-K_{\rm mod,loc}^{\rm pkt}\bigr)
=O(L^{-1+o(1)}),
\qquad r,q\le2.
}
\label{eq:local-model-error}
\end{equation}

By Proposition~\ref{prop:trivial-outer-twist}, the invariant source has no
nontrivial arithmetic whitening. Consequently the exact/model local discrepancy
has normalized diagonal jump $O(L^{-1+o(1)})=o(1)$, and its fixed mixed lag
derivatives are bounded by a fixed power of $L$ on every retained shell.
The shell-local primitive criterion of Lemma~\ref{lem:two-variable-primitive},
together with the metric comparison
\eqref{eq:local-global-Gram-comparison} and the choice
$A_0>C_{\rm reg}+3$ in \eqref{eq:acyclic-A0-choice}, therefore gives
\begin{equation}
\boxed{
\|\Gfr_T^{-1/2}R_T^{\rm loc}\Gfr_T^{-1/2}\|_{S_2}^2=o(N_T).
}
\label{eq:local-HS}
\end{equation}

\begin{definition}[Frozen principal operator]
\label{def:PT}
Let $W_T$ be the stable logarithmic core length fixed in
\eqref{eq:stable-log-core}, and let $C_T$ denote restriction of the straightened
physical logarithmic coordinate to $0\le u\le W_T$.  After this restriction use
the normalized lag coordinate $r=u/L$ and the unitary output rescaling
$g(u)\mapsto L^{1/2}g(Lr)$.  Insert the nonnegative height partition
\eqref{eq:height-principal-partition} linearly before stationary phase.

Define $P_T$ to be the \emph{fully frozen} finite-model principal term
$P_T^{\rm fr}$ of \eqref{eq:common-core-principal-congruence}: after the common
one-sided packet/core analysis is fixed, replace
\[
\lambda(\tau),\qquad \mu_b,\qquad
\rho_{rs}^{(T)}
\]
by
\[
1,\qquad1,\qquad
\rho_{rs}^{(0)}=1-\frac{|w_r-w_s|}{L}
\]
in the sharp completed local profile before taking the normalized curvature
coefficient.  Equivalently,
\begin{equation}
\boxed{
P_T=(\widehat{\mathcal A}_T^{\rm core})^*K_G
\widehat{\mathcal A}_T^{\rm core},
\qquad
K_G(r,s)=G(1-|r-s|).
}
\label{eq:PT-frozen-definition}
\end{equation}
Thus positivity is proved only for this single frozen lag operator.  The actual
finite-$T$ sharp model main is
\[
A_{T,\rm mod}^{\rm sharp}=P_T+R_T^{\rm par}
\]
by Proposition~\ref{prop:common-core-congruence}; the parameter defect
$R_T^{\rm par}$ is assigned to the regular remainder and is never used inside
a variable- or block-parameter positivity argument.

By Proposition~\ref{prop:trivial-outer-twist}, the auxiliary arithmetic map
specializes to its $d=1$ coordinate and every outer-ratio translation is the
identity for the invariant contour source.  No stationary Fourier-mode
restriction is imposed inside the core.  The difference between the exact
smoothed critical-$J$ AFE and its sharp full fibre is likewise excluded from
$P_T$ and belongs to the regular remainder by
\eqref{eq:J-AFE-sharp-error}.

No exact HLP level is included in $P_T$: the whole exact--model residual of
Proposition~\ref{prop:direct-exact-model-residual} remains on the original
absolute-convergence right edge and is treated by direct sidebands.  The
complementary logarithmic tail $W_T<u\le L$ is assigned by
Proposition~\ref{prop:terminal-log-tail}.  The auxiliary outward arithmetic
annulus of Proposition~\ref{prop:outer-excess-edge} is empty in the
invariant-source specialization.  The slow self-freezing discrepancy $f-F_b$
is excluded from $P_T$ and belongs only to the regular remainder.
\end{definition}

\begin{proposition}[Direct exact--model source decomposition on the stable band]
\label{prop:uniform-twisted-replacement}
After the $f'$ source, horizontal connectors, and nonstationary sector have
been separated, restrict the stationary exact Hardy source to the stable
logarithmic band $0\le u\le W_T$. Here ``stable'' refers only to the logarithmic
packet coordinate. By Proposition~\ref{prop:trivial-outer-twist} the invariant
source has no outward arithmetic annulus. On this stable band the same packet contour
matrix has the disjoint decomposition
\begin{equation}
\boxed{
A_{T}^{\rm stat,ex,stable}=P_T+E_T^{\onechi}+R_T^{\rm src}.
}
\label{eq:uniform-twisted-replacement}
\end{equation}
Here $E_T^{\onechi}$ is formed by the low Fourier sidebands of the entire
direct exact--model residual \eqref{eq:direct-exact-model-residual}, together
with the other genuinely direct stationary sources.  Its rank is bounded by
Proposition~\ref{prop:edge-rank}, independently of the residual coefficients or
signs.  Every complementary high-sideband part belongs instead to
$R_T^{\rm src}$.  The same regular source also contains the finite-$T$
parameter defect $R_T^{\rm par}$ of
\eqref{eq:common-core-principal-congruence}, the critical-$J$ smooth-to-sharp
AFE remainder, the slow self-freezing term $f-F_b$, the scalar finite-model HLZ
contour error, and the other mutually exclusive classes identified in
Section~\ref{sec:ledger}.  The auxiliary outer-excess family is zero for the
invariant contour source.
\end{proposition}

\begin{proof}
Work first on the fixed absolute-convergence right edge and separate the
horizontal connectors and the nonstationary sector.  On each retained height
block insert the identity \eqref{eq:direct-model-source-split}.  This is an
exact algebraic source split before any Perron deformation:
\[
\widehat H_{\rm ex}
=H_{\rm mod}+\Delta H_{\rm dir}+(f-F_b).
\]

For the finite model source $H_{\rm mod}$, treat the frozen self coefficient
$F_b$ by the critical-$J$ identity and the logarithmic-derivative faces
$-2P+P(s-2/L)$ by the scalar three-$Z$ calculation.  First apply
Lemma~\ref{lem:HLZ-local-height-rescaling} so that the three-$Z$ Mellin scale is
the same local height $X=\tau/(2\pi)$ as the critical-$J$ source.  Next apply
Lemma~\ref{lem:J-AFE-sharp-transfer}; its smooth-to-sharp discrepancy is retained
in $R_T^{\rm src}$ rather than identified with the sharp fibre.

For the remaining sharp completed model,
Proposition~\ref{prop:common-core-congruence}, with the scalar-HLZ packet lift
of Lemma~\ref{lem:model-HLZ-Fubini-lift}, gives the exact split
\[
A_{T,\rm mod}^{\rm sharp}=P_T+R_T^{\rm par}.
\]
Thus only the fully frozen profile enters $P_T$.  The physical-height parameters
$\lambda(\tau)$, $\mu_b$, and the moving endpoint remain in
$R_T^{\rm par}$ on their original stationary-carrier channel; no
constant-parameter positivity lemma is applied to them.  The relative estimate
for this defect is deferred to Lemma~\ref{lem:finiteT-parameter-regular}.
Proposition~\ref{prop:trivial-outer-twist} shows that no auxiliary outer
arithmetic annulus is present.  The scalar model HLZ contour error is also
retained in $R_T^{\rm src}$; it is not inferred entrywise from a packet matrix.

For the residual $\Delta H_{\rm dir}$ no local projector is applied.  Its safe-line
geometric expansion is absolutely convergent, and
Proposition~\ref{prop:direct-exact-model-residual} keeps every exact HLP level
on the physical total carrier $M$.  Expand the smooth stationary symbol in
Fourier sidebands.  All low sidebands, including the norm-convergent sum over
the complete HLP hierarchy and the translated coefficient shift
$-P(s-2/L)$, have range in the single geometric edge space of
Proposition~\ref{prop:edge-rank}; their sum is $E_T^{\onechi}$ and has rank
$O(T)=o(N_T)$.  The complementary high sidebands retain the coefficient
square-sum and Fourier-decay bounds of
Lemmas~\ref{lem:direct-carrier-square} and \ref{lem:stationary-symbol} and are
assigned only to $R_T^{\rm src}$.

The final term $f-F_b$ is the slow self-freezing discrepancy.  By
Lemma~\ref{lem:block-freezing}, after the common normalization its fixed
packet/shell derivatives are $O(L^{-D-1+C})=o(1)$; it therefore belongs only to
$R_T^{\rm src}$.  The remaining stationary/Stirling corrections and the
smooth height-partition derivatives are disjoint regular terms, while the
terminal logarithmic complement is handled separately by
Proposition~\ref{prop:terminal-log-tail}.  This proves
\eqref{eq:uniform-twisted-replacement} without using the optional exact-HLP
packet-HLZ hypothesis.
\end{proof}
