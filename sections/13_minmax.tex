\section{Finite-dimensional inertia reduction}
\label{sec:minmax}

\begin{lemma}[Relative norms under compression]
\label{lem:relative-compression}
Let $G\succ0$ and let $W$ be a subspace of a finite-dimensional Hermitian
space.  If $R$ is Hermitian, then
\[
\left\|(G|_W)^{-1/2}(R|_W)(G|_W)^{-1/2}\right\|_{S_2}
\le \left\|G^{-1/2}RG^{-1/2}\right\|_{S_2},
\]
and the same inequality holds with $S_2$ replaced by the operator norm.
\end{lemma}

\begin{proof}
Let $J:W\hookrightarrow V$ be the inclusion and set
\[
U:=G^{1/2}J(J^*GJ)^{-1/2}.
\]
Then $U^*U=I_W$, and the whitened compression equals
\[
U^*(G^{-1/2}RG^{-1/2})U.
\]
Both inequalities follow from the ideal property of the Schatten norms and
the fact that $U$ is an isometry.
\end{proof}

\begin{lemma}[One-sided relative-defect deletion]
\label{lem:one-sided-defect-deletion}
Let $A,P,G$ be Hermitian on a finite-dimensional space, with $G\succ0$ and
$P\succeq aG$ for some $a>0$.  Put
\[
D:=G^{-1/2}(A-P)G^{-1/2},
\qquad b:=\tr(D_-^2).
\]
Then there is a subspace $V_\beta$ such that
\begin{equation}
\boxed{
\codim V_\beta\le4a^{-2}b,
\qquad A|_{V_\beta}\succeq\frac a2G|_{V_\beta}.}
\label{eq:one-sided-defect-deletion}
\end{equation}
Only the negative spectrum of $D$ enters this estimate.
\end{lemma}

\begin{proof}
Let $S_\beta$ be the spectral subspace of $D$ corresponding to eigenvalues
strictly below $-a/2$.  Every such eigenvalue contributes more than $a^2/4$
to $\tr(D_-^2)$, and hence
\[
\dim S_\beta\le4a^{-2}b.
\]
On $S_\beta^\perp$ one has $D\succeq-aI/2$.  Pull this subspace back by
$G^{-1/2}$ and call the result $V_\beta$.  Since
$G^{-1/2}PG^{-1/2}\succeq aI$, congruence gives
$A|_{V_\beta}\succeq aG|_{V_\beta}/2$, proving
\eqref{eq:one-sided-defect-deletion}.
\end{proof}

\begin{lemma}[Rank plus relative Hilbert--Schmidt min--max]
\label{lem:minmax}
Let $V$ be finite dimensional. Let $A,P,E,R,G$ be Hermitian on $V$, with
$G\succ0$, and suppose
\[
A=P+E+R,
\qquad
P\succeq c_0G,
\qquad
\rank E\le r,
\]
with
\[
s:=\|G^{-1/2}RG^{-1/2}\|_{S_2}^2.
\]
Then there is a subspace $V_{\rm good}\subset V$ with
\begin{equation}
\boxed{
\codim_VV_{\rm good}\le r+16c_0^{-2}s,
\qquad
A|_{V_{\rm good}}\succeq\frac{3c_0}{4}G|_{V_{\rm good}}.
}
\label{eq:minmax-good-space}
\end{equation}
Consequently
\begin{equation}
\boxed{n_-(A|_V)\le r+16c_0^{-2}s.}
\label{eq:minmax}
\end{equation}
If $V$ has codimension $d$ in the ambient space, then
\begin{equation}
\boxed{n_-(A)\le d+r+16c_0^{-2}s.}
\label{eq:minmax-codim}
\end{equation}
\end{lemma}

\begin{proof}
Whiten by $G$:
\[
\widetilde P=G^{-1/2}PG^{-1/2},
\quad
\widetilde E=G^{-1/2}EG^{-1/2},
\quad
\widetilde R=G^{-1/2}RG^{-1/2}.
\]
Let $S_E=\Ran\widetilde E$. Since $\widetilde E$ is Hermitian,
$S_E^\perp=\ker\widetilde E$, and $\dim S_E\le r$. Let $S_R$ be the spectral
subspace of $\widetilde R$ for $|\lambda|\ge c_0/4$. Then
\[
\frac{c_0^2}{16}\dim S_R\le\|\widetilde R\|_{S_2}^2=s,
\]
so $\dim S_R\le16c_0^{-2}s$. Put
\[
\widetilde V_{\rm good}:=(S_E+S_R)^\perp,
\qquad
V_{\rm good}:=G^{-1/2}\widetilde V_{\rm good}.
\]
On this space,
\[
\widetilde P+\widetilde E+\widetilde R
\succeq c_0I-\frac{c_0}{4}I
=\frac{3c_0}{4}I.
\]
This proves \eqref{eq:minmax-good-space}. Since $A$ is positive on
$V_{\rm good}$, every negative subspace of $A|_V$ meets $V_{\rm good}$
trivially; hence its dimension is at most
$\codim_VV_{\rm good}\le r+16c_0^{-2}s$, proving \eqref{eq:minmax}.
If $V$ has ambient codimension $d$, the dimension inequality for the
intersection of a maximal negative subspace with $V$ gives
\eqref{eq:minmax-codim}.
\end{proof}
