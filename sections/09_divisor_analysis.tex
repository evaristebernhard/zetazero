\section{Auxiliary finite divisor comparison}
\label{sec:divisor-gram}

The divisor maps in this section are finite auxiliary comparison coordinates. They are
not, by themselves, a truncation of the exact outer-divisor source and are not
replaced by a scalar normalization.

\begin{lemma}[Hilbert-valued gcd identity]
\label{lem:gcd-identity}
For $b_1,\dots,b_Y$ in a Hilbert space $\mathcal H$,
\begin{equation}
\boxed{
\sum_{m,n\le Y}\frac{(m,n)}{mn}\langle b_m,b_n\rangle
=
\sum_{d\le Y}\varphi(d)
\left\|\sum_{d\mid n}\frac{b_n}{n}\right\|^2
\ge0.
}
\label{eq:gcd-identity}
\end{equation}
\end{lemma}

\begin{proof}
Insert the elementary identity
\[
(m,n)=\sum_{d\mid(m,n)}\varphi(d)
\]
and interchange the finite sums. The inner double sum becomes the squared
Hilbert norm in \eqref{eq:gcd-identity}.
\end{proof}

Let $U_n$ be isometries on $\mathcal H$ and define
\begin{equation}
(D_Yb)_d:=\sqrt{\varphi(d)}
\sum_{d\mid n}\frac{U_nb_n}{n},
\qquad d\le Y.
\label{eq:divisor-map}
\end{equation}
Thus the auxiliary finite arithmetic Gram is
\begin{equation}
\boxed{G_{\rm ar}:=D_Y^*D_Y.}
\label{eq:arith-gram}
\end{equation}
The isometries $U_n$ may contain the exact shell translations coming from the
physical one-$\chi$ carrier; no scalarization is involved.

The common outer-divisor feature of
Proposition~\ref{prop:common-divisor-feature} has the normalized analysis map
\begin{equation}
(\widetilde{\mathcal W}_ac)(N)
:=N^{-1/2-a}\sum_{\substack{d\mid N\\d\le Y}}c_dd^a.
\label{eq:normalized-common-divisor-map}
\end{equation}
Its full output contains, in particular, the coordinates $1\le N\le Y$.
Writing $S_N=N^{1/2+a}(\widetilde{\mathcal W}_ac)(N)$, finite M\"obius
inversion gives
\begin{equation}
\boxed{c_nn^a=\sum_{k\mid n}\mu(k)S_{n/k}.}
\label{eq:common-divisor-mobius}
\end{equation}
For bounded $p$ and $a=p/L$, $n^{\pm a}=L^{o(1)}$ uniformly for $n\le Y$.
Cauchy--Schwarz in \eqref{eq:common-divisor-mobius} therefore gives the
conservative inverse estimate
\begin{equation}
\boxed{
\|\widetilde{\mathcal W}_a^{-1}\|_{\op,\Ran\widetilde{\mathcal W}_a}
\le Y^{1+o(1)}.
}
\label{eq:common-divisor-inverse}
\end{equation}
Thus the new unsummed factorization introduces no larger arithmetic inverse
loss than the finite divisor maps below.

\begin{proposition}[Finite M\"obius inverse]
\label{prop:divisor-inverse}
The map $D_Y$ is injective and
\begin{equation}
\boxed{
\|D_Y^{-1}\|_{\op,\Ran D_Y}
\le Y\tau_{\max}(Y)^{1/2}=Y^{1+o(1)}.
}
\label{eq:divisor-inverse}
\end{equation}
The estimate is uniform in the ambient Hilbert space.
\end{proposition}

\begin{proof}
Put
\[
c_n:=\frac{U_nb_n}{n},
\qquad
S_d:=\frac{(D_Yb)_d}{\sqrt{\varphi(d)}}.
\]
Then
\[
S_d=\sum_{d\mid n}c_n.
\]
Finite M\"obius inversion on the divisibility poset gives
\begin{equation}
\boxed{
c_n=\sum_{k\le Y/n}\mu(k)S_{nk}.}
\label{eq:mobius-inverse}
\end{equation}
Since $U_n$ is an isometry,
\begin{align*}
\|b_n\|
&\le n\sum_{k\le Y/n}
\frac{\|(D_Yb)_{nk}\|}{\sqrt{\varphi(nk)}}\\
&\le \sqrt{nY}
\left(\sum_{k\le Y/n}\|(D_Yb)_{nk}\|^2\right)^{1/2}.
\end{align*}
Squaring and summing in $n$ yields
\[
\sum_{n\le Y}\|b_n\|^2
\le Y^2\tau_{\max}(Y)\|D_Yb\|^2,
\]
which proves \eqref{eq:divisor-inverse}. Injectivity follows either from this
bound or directly by descending induction in $d$.
\end{proof}

\begin{lemma}[Conservative arithmetic transfer loss]
\label{lem:arithmetic-loss}
For an auxiliary finite twisted family with
$1\le h,k,g\le Y=L^\kappa$ and fixed $\kappa>0$, the arithmetic label space
has dimension
\begin{equation}
\boxed{\dim\mathcal H_{\rm ar}\le Y^{3+o(1)}.}
\label{eq:arithmetic-dimension}
\end{equation}
Every retained finite arithmetic pullback used in the small-outer comparison below
contains at most three nonunitary divisibility maps in total; each is either the common map
\eqref{eq:normalized-common-divisor-map} or a finite auxiliary map of the form
\eqref{eq:divisor-map}. Hence the inverse of one complete arithmetic analysis
map on its range has norm at most
\begin{equation}
\boxed{Y^{3+o(1)}.}
\label{eq:arithmetic-loss}
\end{equation}
Moreover, any Hermitian arithmetic-fibre kernel whose entries, after the common
positive weights have been factored out, have operator norm at most
$\varepsilon$ transfers to the divisor-whitened metric with the deliberately
coarse bound
\begin{equation}
\boxed{
\|G_{\rm ar}^{-1/2}R_{\rm ar}G_{\rm ar}^{-1/2}\|_{\op}
\ll \varepsilon Y^{10+o(1)}.
}
\label{eq:arithmetic-two-sided-transfer}
\end{equation}
The exponent $10$ is only a safety exponent.
For the Hilbert--Schmidt norm on the same arithmetic fibre one also has
\begin{equation}
\boxed{
\|G_{\rm ar}^{-1/2}R_{\rm ar}G_{\rm ar}^{-1/2}\|_{S_2}
\ll \varepsilon Y^{23/2+o(1)},
}
\label{eq:arithmetic-two-sided-HS-transfer}
\end{equation}
by \eqref{eq:arithmetic-dimension}. In particular a polylogarithmic
entrywise mixed-derivative bound remains polylogarithmic after arithmetic
whitening.
\end{lemma}

\begin{proof}
There are at most $Y^3$ triples $(g,h,k)$, with divisor refinements contributing
only $Y^{o(1)}$. Each nonunitary divisibility inverse has norm
$Y^{1+o(1)}$ by \eqref{eq:common-divisor-inverse} or
Proposition~\ref{prop:divisor-inverse}; three factors give
\eqref{eq:arithmetic-loss}.

For the two-sided estimate, first take the crude operator norm of a normalized
arithmetic matrix of entry size at most $\varepsilon$. By
\eqref{eq:arithmetic-dimension} it is at most
$\varepsilon Y^{3+o(1)}$. Whitening on the left and right costs at most one
complete inverse factor \eqref{eq:arithmetic-loss} on each side, hence at most
$Y^{6+o(1)}$. Thus the total loss is
$\varepsilon Y^{9+o(1)}$, which is bounded by the safer displayed
$\varepsilon Y^{10+o(1)}$ for all large $Y$. Perturbations that factor through
the common analysis map, or arise from a one-sided deformation of that map,
actually cost less; throughout the proof we are free to use the uniform
worst-case bound \eqref{eq:arithmetic-two-sided-transfer}. Finally, on an
arithmetic fibre of dimension at most $Y^{3+o(1)}$,
\[
\|B\|_{S_2}\le(\dim\mathcal H_{\rm ar})^{1/2}\|B\|_{\op},
\]
so \eqref{eq:arithmetic-two-sided-transfer} implies
\eqref{eq:arithmetic-two-sided-HS-transfer}.
\end{proof}

\begin{lemma}[Operator-valued positive lift]
\label{lem:operator-gram}
If $(H_{mn})\succeq0$ and $U_n$ are unitary, then for arbitrary scalars $a_n$,
\[
\sum_{m,n}\overline{a_m}a_nH_{mn}U_m^*U_n\succeq0.
\]
\end{lemma}

\begin{proof}
Choose a Gram representation
$H_{mn}=\sum_j\overline{x_{j,m}}x_{j,n}$ and set
$X_j=\sum_na_nx_{j,n}U_n$. The displayed operator is
$\sum_jX_j^*X_j$.
\end{proof}
