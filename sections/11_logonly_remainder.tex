\section{Log-only remainder ledger}
\label{sec:logonly-remainder-ledger}

This is the remainder ledger used by the main theorem.  Every theorem-facing
regular class is measured with the absolute-lag primitive criterion of
Section~\ref{sec:primitive-remainder-core}.  In particular, no one-sided
compressed-shift representation of the packet-polarized logarithmic source is
used.

\begin{lemma}[Lagwise HLZ contour error is relative-regular]
\label{lem:lagwise-HLZ-error-regular}
Let $R_{\log,T}^{\rm HLZ}$ be the error in
Proposition~\ref{prop:lagwise-packet-HLZ} after the finite Cauchy residues and
curvature extraction. On every retained compression of dimension $O(N_T)$,
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{\log,T}^{\rm HLZ}\Gfr_T^{-1/2}\right\|_{S_2}^2
=o(N_T).
}
\label{eq:lagwise-HLZ-relative-HS}
\end{equation}
\end{lemma}

\begin{proof}
Let $\varepsilon_T(\rho)$ denote any scalar error profile produced after the
separated Cauchy residues.  By
Lemma~\ref{lem:scalar-lagweight-HLZ} and
Proposition~\ref{prop:lagwise-packet-HLZ}, for every prescribed $A>0$,
\begin{equation}
\max_{0\le j\le2}\sup_\rho
|\partial_\rho^j\varepsilon_T(\rho)|
\ll L^{-A+C_0}
\label{eq:lagwise-HLZ-profile-saving}
\end{equation}
for one fixed loss $C_0$ after the finite Cauchy and curvature operations.

On a unit physical shell $x=J_j+t$, $y=J_j+s$, the corresponding normalized
kernel is a finite sum of terms
\begin{equation}
D_{j,T}^{\rm HLZ}(t,s)
=\frac1L a_{j,T}(t,s)
\varepsilon_T\!\left(
\nu_{j,T}(t,s)-
\frac{|\log(J_j+t)-\log(J_j+s)|}{L}
\right),
\label{eq:lagwise-HLZ-shell-kernel}
\end{equation}
where $|\nu_{j,T}-1|\ll L^{-1}$ and the smooth amplitude and its first two
shell derivatives have fixed polylogarithmic bounds by
Lemma~\ref{lem:stationary-symbol} and the bounded-overlap height partition.

Apply Lemma~\ref{lem:logonly-shell-normalization-transfer}.  The normalized
diagonal jump is $O(L^{-A+C_1})=o(1)$, while the per-shell squared
Hilbert--Schmidt mass of the normalized regular mixed derivative is
$O(L^{-2A+C_2})$.  Since the number of retained shells is
$O(X_T)=O(TL^{-A_0})$, the arbitrary scalar saving may be chosen so that the
shell sum is $o(N_T)$.  Lemma~\ref{lem:logonly-two-variable-primitive} proves
\eqref{eq:lagwise-HLZ-relative-HS}.
\end{proof}

\begin{lemma}[Finite-height log-profile defect is relative-regular]
\label{lem:logonly-finite-height-regular}
Let
\[
\Delta_T^{\rm par}(d;p)
:=\mathcal G^{\log}_{\lambda(\tau),\lambda(\tau)-|d|/L}(p)
 -\mathcal G^{\log}_{1,1-|d|/L}(p).
\]
After the fixed curvature extraction, let $R_{\log,T}^{\rm par}$ be the
corresponding packet form. Then
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{\log,T}^{\rm par}\Gfr_T^{-1/2}\right\|_{S_2}^2
=o(N_T).
}
\label{eq:logonly-finite-height-relative-HS}
\end{equation}
\end{lemma}

\begin{proof}
For bounded $p$ the function
$(\lambda,\rho,p)\mapsto\mathcal G^{\log}_{\lambda,\rho}(p)$ is analytic on a
fixed neighbourhood of the retained parameter set.  Since
$\lambda(\tau)=1+O(L^{-1})$, the mean-value theorem gives, through two
$\rho$- and curvature derivatives,
\begin{equation}
\max_{0\le r\le2}\sup_d
|\partial_\rho^r\Delta_T^{\rm par}(d;p)|
\ll L^{-1+C}.
\label{eq:logonly-finite-height-profile-bound}
\end{equation}
The shell kernel is therefore of the form covered by
Lemma~\ref{lem:logonly-shell-normalization-transfer}.  Its normalized jump is
$o(1)$ after the fixed hierarchy is chosen, and the normalized regular mixed
derivative has a fixed polylogarithmic per-shell squared Hilbert--Schmidt
bound.  Taking $A_0$ larger than this fixed exponent makes the shell sum
$o(N_T)$.  Lemma~\ref{lem:logonly-two-variable-primitive} gives
\eqref{eq:logonly-finite-height-relative-HS}.
\end{proof}

\begin{lemma}[The exact direct residual is relative-regular]
\label{lem:deltaH-dir-relative-regular}
Let $R_{\rm dir,T}$ be the right-edge contribution of
$\mathcal C\Delta H_{\rm dir}$ in \eqref{eq:logonly-product-split}, including
its adjoint orientation and the fixed stationary, height-block, and curvature
normalizations. Then
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{\rm dir,T}\Gfr_T^{-1/2}\right\|_{S_2}^2=o(N_T).
}
\label{eq:deltaH-dir-relative-HS}
\end{equation}
No low-rank compressed-shift decomposition is used.
\end{lemma}

\begin{proof}
Lemma~\ref{lem:deltaH-dir-small} gives
$\Delta H_{\rm dir}=O(L^{-2})$ on the safe line through all fixed source
derivatives, up to a fixed polylogarithmic factor.  Keeping the exact
stationary lag geometry, the shell-local normalized kernel is a finite sum of
terms
\begin{equation}
D_{j,T}^{\rm dir}(t,s)
=\frac1L a_{j,T}(t,s)
H_{j,T}^{\rm dir}\!\left(
\nu_{j,T}(t,s)-
\frac{|\log(J_j+t)-\log(J_j+s)|}{L}
\right),
\label{eq:deltaH-dir-shell-kernel}
\end{equation}
with
\begin{equation}
\max_{0\le r\le2}\sup_\rho
|\partial_\rho^rH_{j,T}^{\rm dir}(\rho)|
\ll L^{-2+C_{\rm dir,reg}}.
\label{eq:deltaH-dir-profile-bound}
\end{equation}
Lemma~\ref{lem:logonly-shell-normalization-transfer} gives the normalized jump
and mixed-derivative estimates, and
Lemma~\ref{lem:logonly-two-variable-primitive} converts them into the relative
Hilbert--Schmidt bound.  Choosing $A_0$ after the fixed source exponent makes
the shell sum $o(N_T)$.
\end{proof}

\begin{lemma}[Log-only $f'$ class is relative-regular]
\label{lem:logonly-fprime-relative-regular}
Let $R_{f',T}^{\log}$ be the contribution of
\eqref{eq:logonly-fprime-source}. Then
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_{f',T}^{\log}\Gfr_T^{-1/2}\right\|_{S_2}^2=o(N_T).
}
\label{eq:logonly-fprime-relative-HS}
\end{equation}
\end{lemma}

\begin{proof}
Every term in \eqref{eq:logonly-fprime-source} contains at least one factor
$f'=O(T^{-1})$ on the retained height strip, while
$(f-P)^{-1}=O(L^{-1})$ on the fixed safe line.  All fixed stationary, shell,
curvature and Cauchy derivatives cost only a fixed power of $L$.  Hence each
shell-local profile and its first two lag derivatives are
\[
O(T^{-1}L^C).
\]
Lemma~\ref{lem:logonly-shell-normalization-transfer} therefore gives a
normalized jump $O(T^{-1}L^C)=o(1)$ and a per-shell normalized mixed-derivative
squared Hilbert--Schmidt mass $O(T^{-2}L^C)$.  Summing over
$O(TL^{-A_0})$ shells is $o(N_T)$ for every sufficiently large fixed $A_0$.
Lemma~\ref{lem:logonly-two-variable-primitive} proves the assertion.
\end{proof}

\begin{proposition}[Principal plus remainder decomposition]
\label{prop:logonly-final-decomposition}
On the retained space $\Pret$,
\begin{equation}
\boxed{A_T=P_T+E_T^{\rm geom}+R_T^{\log},}
\label{eq:logonly-final-decomposition}
\end{equation}
where $P_T=A_T^{G_{\log}}$ is the log-only principal operator of
Section~\ref{sec:model-floor},
\begin{equation}
\rank E_T^{\rm geom}=o(N_T),
\label{eq:logonly-edge-rank}
\end{equation}
and
\begin{equation}
\boxed{
\left\|\Gfr_T^{-1/2}R_T^{\log}\Gfr_T^{-1/2}\right\|_{S_2}^2=o(N_T).
}
\label{eq:logonly-total-relative-HS}
\end{equation}
The edge operator contains only genuinely geometric stationary-buffer,
exterior-packet, horizontal-collar, and source-independent finite-rank
families. It contains no part of $H_{\log}$.
\end{proposition}

\begin{proof}
Lemma~\ref{lem:holomorphic-self-elimination} removes the critical-$J$ self
block before principal reduction.  Proposition~\ref{prop:lagwise-packet-HLZ}
and Corollary~\ref{cor:frozen-logonly-lag-kernel} identify the unique order-one
principal source with $P_T=A_T^{G_{\log}}$.
Lemma~\ref{lem:logonly-finite-height-regular} handles the finite-height defect;
Lemma~\ref{lem:lagwise-HLZ-error-regular} handles the scalar HLZ contour error;
Lemma~\ref{lem:deltaH-dir-relative-regular} handles the exact direct residual;
and Lemma~\ref{lem:logonly-fprime-relative-regular} handles the $f'$ family.
The higher stationary/Stirling corrections and nonstationary leakage are
source-independent analytic remainder mechanisms from
Sections~\ref{sec:stationary} and \ref{sec:shell-frame}; their existing
integration-by-parts estimates are unchanged by replacing the scalar principal
profile.

There is no critical-$J$ AFE smooth-to-sharp remainder, no frozen-self class,
and no low/high sideband decomposition of the logarithmic principal source.
The remaining finite-rank pieces are the genuinely geometric edge families,
whose total rank is $o(N_T)$.  Summing the relative-Hilbert--Schmidt classes
proves \eqref{eq:logonly-total-relative-HS}.
\end{proof}

\begin{remark}[Scope of compressed shifts]
Compressed Fourier shifts may still be used for genuinely one-sided exterior
or auxiliary terms for which the physical variable is itself one-sided.  They
are not part of the packet-polarized logarithmic principal source or of the
three theorem-facing regular classes above.
\end{remark}
